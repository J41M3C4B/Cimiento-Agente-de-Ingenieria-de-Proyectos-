//! Reading a call whole, once, into a fixed sheet (ADR-014, option 3: "lectura completa estructurada").
//!
//! The rail (ADR-013) cuts the call into fragments with rules written for the formats of one funder
//! and asks the model only to label them. This does the opposite: ONE call hands the model every page
//! of every document of the call and asks it to fill a fixed sheet of sections, each item a quote
//! copied from a page. The code verifies every quote against the page it cites and keeps the quote as
//! the item's text, so a paraphrase or an invented item cannot get in; dates and limits are read from
//! the quote by the same readers the extractor uses, and a limit whose number is not in its quote is
//! thrown away. There is no step that decides whether a document "is a call": if there is nothing to
//! fill, the sheet is empty.
//!
//! It needs no rule written for a format, which is the point: what changes from one funder to the
//! next is the model's job, and what stays (verification, readers, closed lists) is the code's.

// Not wired to a command yet: tests and the live comparison use it.

/// How much text goes to the model, in characters (about 40,000 tokens). A call is 15,000 to 50,000
/// characters; this only bites on a very long bundle, and what is left out is reported.
pub const SHEET_BUDGET_CHARS: usize = 160_000;
/// A quote shorter than this proves nothing.
const MIN_QUOTE_CHARS: usize = 12;

/// Sections whose items are a verified quote and its page.
pub const SECTIONS: [&str; 10] = [
    "funder",
    "objective",
    "eligibility",
    "exclusions",
    "documents",
    "evaluation",
    "project_requirements",
    "funding_conditions",
    "population",
    "funded",
];
pub const DATE_KINDS: [&str; 6] = ["application_window", "registration", "info_session", "results", "disbursement", "other"];
pub const LIMIT_KINDS: [&str; 7] = [
    "max_duration_months",
    "max_admin_percent",
    "min_operating_years",
    "max_amount",
    "min_amount",
    "counterpart_percent",
    "other",
];

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct SheetItem {
    /// The quote, verified to be in the page: the PDF's own words.
    pub text: String,
    pub page: usize,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct SheetDate {
    pub kind: &'static str,
    pub text: String,
    pub page: usize,
    /// What the date readers found in the quote (`2026-06-22`, `2026-07-15`…); empty if none.
    pub read: Vec<String>,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct SheetLimit {
    pub kind: &'static str,
    /// Read from the quote, or the model's number when it appears in the quote.
    pub value: f64,
    pub text: String,
    pub page: usize,
}

#[derive(Debug, Clone, Default, PartialEq, Serialize)]
pub struct CallSheet {
    pub sections: BTreeMap<&'static str, Vec<SheetItem>>,
    pub dates: Vec<SheetDate>,
    pub limits: Vec<SheetLimit>,
    /// Contradictions and ambiguities the model reports, as it wrote them: its opinion, not the call's text.
    pub doubts: Vec<String>,
}

impl CallSheet {
    pub fn items(&self) -> usize {
        self.sections.values().map(Vec::len).sum::<usize>() + self.dates.len() + self.limits.len()
    }

    /// Every text of the sheet, for comparing it with a reference.
    pub fn all_text(&self) -> String {
        let mut out: Vec<&str> = Vec::new();
        out.extend(self.sections.values().flatten().map(|i| i.text.as_str()));
        out.extend(self.dates.iter().map(|d| d.text.as_str()));
        out.extend(self.limits.iter().map(|l| l.text.as_str()));
        out.extend(self.doubts.iter().map(String::as_str));
        out.join("\n")
    }
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct SheetReport {
    /// `used`, `not_needed`, or the kind of failure (`offline`, `quota_reached`…).
    pub ai: String,
    pub pages_sent: usize,
    /// Pages left out because the bundle is longer than the budget.
    pub pages_dropped: usize,
    pub chars_sent: usize,
    pub kept: usize,
    /// Items thrown away: a quote that is not in the page, a number not in its quote, an unknown kind.
    pub rejected: Vec<Rejected>,
    /// Items that said again what another of the same section already said.
    pub duplicates: usize,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct SheetDraft {
    pub sheet: CallSheet,
    pub report: SheetReport,
}

impl SheetReport {
    fn without_ai(ai: &str) -> Self {
        SheetReport { ai: ai.into(), pages_sent: 0, pages_dropped: 0, chars_sent: 0, kept: 0, rejected: Vec::new(), duplicates: 0 }
    }
}

// ------------------------------------------------------------------ verifying

macro_rules! re {
    ($p:expr) => {{
        static R: OnceLock<Regex> = OnceLock::new();
        R.get_or_init(|| Regex::new($p).expect("valid regex"))
    }};
}

/// The numbers written in a quote: "$300,000.00" is 300000, "10 %" is 10, "tres" is not a number.
fn numbers_in(text: &str) -> Vec<f64> {
    re!(r"\d[\d,. ]*\d|\d")
        .find_iter(text)
        .filter_map(|m| m.as_str().replace([',', ' '], "").trim_end_matches('.').parse::<f64>().ok())
        .collect()
}

struct PageIndex {
    norm: BTreeMap<usize, String>,
}

impl PageIndex {
    fn new(pages: &[PageText]) -> Self {
        PageIndex { norm: pages.iter().map(|p| (p.number, norm(&clean_text(&p.text)))).collect() }
    }

    /// `Ok` when the quote is in the page it cites, whatever the line breaks, capitals or spacing.
    fn verify(&self, quote: &str, page: usize) -> Result<(), String> {
        let q = norm(quote);
        if q.chars().count() < MIN_QUOTE_CHARS {
            return Err("la cita es demasiado corta o está vacía".into());
        }
        match self.norm.get(&page) {
            None => Err(format!("la página {page} no existe")),
            Some(text) if text.contains(&q) => Ok(()),
            Some(_) => Err(format!("la cita no aparece en la página {page}")),
        }
    }
}

fn str_of<'a>(v: &'a Value, key: &str) -> &'a str {
    v[key].as_str().unwrap_or("").trim()
}

fn page_of(v: &Value) -> usize {
    v["page"].as_u64().unwrap_or(0) as usize
}

// ------------------------------------------------------------------ applying the answer

/// Builds the sheet from the model's answer. Pure and deterministic: the same pages and the same
/// answer always give the same sheet. `year` is the call's own year, for dates written without one.
pub fn apply_sheet(pages: &[PageText], year: Option<u32>, answer: &Value) -> SheetDraft {
    let index = PageIndex::new(pages);
    let mut report = SheetReport::without_ai("used");
    report.pages_sent = pages.len();
    let mut sheet = CallSheet::default();
    let reject = |report: &mut SheetReport, what: String, why: String| report.rejected.push(Rejected { what, why });
    let short = |s: &str| s.chars().take(60).collect::<String>();

    for section in SECTIONS {
        let mut kept: Vec<SheetItem> = Vec::new();
        let mut known: Vec<String> = Vec::new();
        for item in answer[section].as_array().into_iter().flatten() {
            let (quote, page) = (str_of(item, "quote"), page_of(item));
            let what = format!("{section}: «{}»", short(quote));
            if let Err(why) = index.verify(quote, page) {
                reject(&mut report, what, why);
                continue;
            }
            let n = norm(quote);
            if known.iter().any(|k| same_statement(k, &n)) {
                report.duplicates += 1;
                continue;
            }
            known.push(n);
            kept.push(SheetItem { text: quote.to_string(), page });
        }
        sheet.sections.insert(section, kept);
    }

    let mut known_dates: Vec<String> = Vec::new();
    for d in answer["dates"].as_array().into_iter().flatten() {
        let (quote, page) = (str_of(d, "quote"), page_of(d));
        let what = format!("fecha: «{}»", short(quote));
        let Some(kind) = DATE_KINDS.iter().find(|k| **k == str_of(d, "kind")) else {
            reject(&mut report, what, "el tipo de fecha no está en la lista permitida".into());
            continue;
        };
        if let Err(why) = index.verify(quote, page) {
            reject(&mut report, what, why);
            continue;
        }
        let n = norm(quote);
        if known_dates.iter().any(|k| same_statement(k, &n)) {
            report.duplicates += 1;
            continue;
        }
        known_dates.push(n);
        let read: Vec<String> = dates_in_loose(quote, page, year)
            .into_iter()
            .flat_map(|x| std::iter::once(x.start).chain(x.end))
            .collect();
        sheet.dates.push(SheetDate { kind, text: quote.to_string(), page, read });
    }

    for l in answer["limits"].as_array().into_iter().flatten() {
        let (quote, page) = (str_of(l, "quote"), page_of(l));
        let what = format!("tope: «{}»", short(quote));
        let Some(kind) = LIMIT_KINDS.iter().find(|k| **k == str_of(l, "kind")) else {
            reject(&mut report, what, "el tipo de tope no está en la lista permitida".into());
            continue;
        };
        if let Err(why) = index.verify(quote, page) {
            reject(&mut report, what, why);
            continue;
        }
        // the value comes from the quote: read by the extractor's readers, or the model's own number
        // only if it is written in the quote
        let said = l["value"].as_f64().unwrap_or(f64::NAN);
        let read = limits_in(quote, page).into_iter().find(|x| x.kind == *kind).map(|x| x.value);
        let value = match read {
            Some(v) => v,
            None if numbers_in(quote).iter().any(|n| (n - said).abs() < 1e-9) => said,
            None => {
                reject(&mut report, what, format!("el número {said} no está escrito en la cita"));
                continue;
            }
        };
        if sheet.limits.iter().any(|x| x.kind == *kind && (x.value - value).abs() < 1e-9 && same_statement(&norm(&x.text), &norm(quote))) {
            report.duplicates += 1;
            continue;
        }
        sheet.limits.push(SheetLimit { kind, value, text: quote.to_string(), page });
    }

    sheet.doubts = answer["doubts"]
        .as_array()
        .into_iter()
        .flatten()
        .filter_map(|d| d.as_str())
        .map(|d| d.trim().chars().take(300).collect::<String>())
        .filter(|d| !d.is_empty())
        .take(15)
        .collect();

    report.kept = sheet.items();
    SheetDraft { sheet, report }
}

// ------------------------------------------------------------------ the reading itself

/// The pages as the model sees them, within the budget; how many were left out.
fn context_of(pages: &[PageText], budget: usize) -> (String, usize, usize) {
    let mut out = String::new();
    let mut sent = 0;
    for p in pages {
        let block = format!("Página {}:\n{}\n\n", p.number, clean_text(&p.text));
        if out.chars().count() + block.chars().count() > budget {
            break;
        }
        out.push_str(&block);
        sent += 1;
    }
    (out.trim_end().to_string(), sent, pages.len() - sent)
}

/// Reads the whole call with the model, once, and builds its sheet. Never fails: if the reading
/// cannot be done (no key, no internet, no allowance) the sheet is empty and the report says why.
pub async fn read_call_sheet(
    provider: &dyn AiProvider,
    scanner: &dyn SensitiveScanner,
    ledger: &dyn Ledger,
    project_id: Option<String>,
    pages: &[PageText],
) -> SheetDraft {
    let skip = |ai: &str| SheetDraft { sheet: CallSheet::default(), report: SheetReport::without_ai(ai) };
    if pages.iter().all(|p| p.text.trim().is_empty()) {
        return skip("not_needed");
    }
    let (context, sent, dropped) = context_of(pages, SHEET_BUDGET_CHARS);
    let kept: Vec<PageText> = pages.iter().take(sent).cloned().collect();
    let call = AiCall {
        task: AiTask::CallSheet,
        context: vec![context.clone()],
        user: "Lea la convocatoria completa y llene la ficha.".into(),
        project_id,
    };
    match pipeline::run(provider, scanner, ledger, call).await {
        Ok(answer) => {
            let mut draft = apply_sheet(&kept, extract_call(pages).year, &answer);
            draft.report.pages_dropped = dropped;
            draft.report.chars_sent = context.chars().count();
            draft
        }
        Err(e) => skip(&e.kind()),
    }
}

        assert!(!v.is_valid(&missing), "every section is required, an empty list is how it says 'nothing'");
    }
