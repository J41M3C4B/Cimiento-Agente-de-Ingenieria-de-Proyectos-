//! Second stage of reading a call: the labelling of the rail (ADR-013).
//!
//! The first stage (`call_extract`) reads what rules can read. The rail then cuts the call into
//! candidates (`call_rail`), and ONE model call labels each of them from a closed vocabulary and
//! answers a fixed list of questions with the ids of the candidates that state each thing. The
//! model never writes the call's text: it picks ids and words from closed lists, and the code puts
//! the PDF's own text where each one goes. That is what makes a cheap model and an expensive one
//! give the same result, and what makes a paraphrase or an invented quote impossible.
//!
//! What the model adds is marked as an AI assumption until a person confirms it. A model that
//! disagrees with the rules never overwrites them: the disagreement is listed.

macro_rules! re {
    ($p:expr) => {{
        static R: OnceLock<Regex> = OnceLock::new();
        R.get_or_init(|| Regex::new($p).expect("valid regex"))
    }};
}

/// How much candidate text is sent to the model, in characters (about 8,000 tokens). A real call
/// is 14,000 to 21,000 characters of candidates, so this only bites on very long documents.
pub const LABEL_BUDGET_CHARS: usize = 30_000;

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct ProfileItem {
    /// The PDF's own text (a candidate), never the model's wording.
    pub text: String,
    /// `exclusion`, `priority`, `evaluation`, `objective` or `funder`.
    pub label: &'static str,
    pub applies_to: &'static str,
    pub evidence: Evidence,
    /// Id of the candidate it came from.
    pub candidate: String,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct AgendaItem {
    pub name: String,
    pub evidence: Evidence,
    pub candidate: String,
}

/// What a person needs to know to decide whether a project fits the call.
#[derive(Debug, Clone, Default, PartialEq, Serialize)]
pub struct CallProfile {
    pub funder: Option<ProfileItem>,
    pub objective: Option<ProfileItem>,
    pub agendas: Vec<AgendaItem>,
    pub evaluation: Vec<ProfileItem>,
    pub restrictions: Vec<ProfileItem>,
    /// What the project itself must include or show (not a requirement of the organization).
    pub project_requirements: Vec<ProfileItem>,
    /// What the organization must or must not do if it is funded.
    pub funding_conditions: Vec<ProfileItem>,
    /// Whom the call or each program serves, and whom it does not (ages, groups, type of institution).
    pub population: Vec<ProfileItem>,
    /// What can be funded.
    pub funded: Vec<ProfileItem>,
    /// The stages of the process with their dates, as the call writes them.
    pub calendar: Vec<ProfileItem>,
    /// How, where, when and in what format the application and documents are delivered.
    pub submission: Vec<ProfileItem>,
    pub doubts: Vec<String>,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct Rejected {
    pub what: String,
    pub why: String,
}

/// What came out of one of the fixed questions.
#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct FieldReport {
    pub slot: String,
    /// `found`, `not_in_text`, `unanswered` (the model skipped it) or `unreadable` (it named
    /// candidates but no value could be read from them).
    pub status: String,
    pub ids: Vec<String>,
    /// What the code read from those candidates.
    pub read: Vec<String>,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct ReviewReport {
    /// `used`, `not_needed`, or the kind of failure (`offline`, `quota_reached`…): a failed
    /// review leaves the rules' extraction as it was.
    pub ai: String,
    pub candidates_sent: usize,
    /// Candidates left out because the document is longer than the budget (weakest signals first).
    pub candidates_dropped: usize,
    pub pages_sent: Vec<usize>,
    pub chars_sent: usize,
    pub labeled: usize,
    /// Ids the model never labeled, even after being asked again. A person looks at these.
    pub unlabeled: Vec<String>,
    /// The model skipped some ids, so it was asked once more about only those.
    pub asked_again: bool,
    pub by_label: BTreeMap<String, usize>,
    /// The label and `applies_to` each candidate ended with (after the contract of each label).
    /// What two models are compared on.
    pub labels_by_id: BTreeMap<String, (String, String)>,
    pub fields: Vec<FieldReport>,
    /// Where the model and the rules do not agree. The rules stand; a person decides.
    pub disagreements: Vec<String>,
    pub applied: Vec<String>,
    pub rejected: Vec<Rejected>,
}

impl ReviewReport {
    fn without_ai(ai: &str) -> Self {
        ReviewReport {
            ai: ai.into(),
            candidates_sent: 0,
            candidates_dropped: 0,
            pages_sent: Vec::new(),
            chars_sent: 0,
            labeled: 0,
            unlabeled: Vec::new(),
            asked_again: false,
            by_label: BTreeMap::new(),
            labels_by_id: BTreeMap::new(),
            fields: Vec::new(),
            disagreements: Vec::new(),
            applied: Vec::new(),
            rejected: Vec::new(),
        }
    }
}

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct CallDraft {
    pub extraction: CallExtraction,
    pub profile: CallProfile,
    pub report: ReviewReport,
}

// ------------------------------------------------------------------ comparing texts

/// Lowercase letters and digits separated by single spaces.
pub(crate) fn norm(s: &str) -> String {
    let mut out = String::with_capacity(s.len());
    let mut last_space = true;
    for ch in s.chars() {
        if ch.is_alphanumeric() {
            out.extend(ch.to_lowercase());
            last_space = false;
        } else if !last_space {
            out.push(' ');
            last_space = true;
        }
    }
    out.trim().to_string()
}

/// Two normalized texts say the same thing: one contains the other, or most of their words are
/// shared. Different figures always make them different ("12 meses" is not "10 meses").
pub(crate) fn same_statement(a: &str, b: &str) -> bool {
    let words = |s: &str| s.split_whitespace().map(str::to_string).collect::<BTreeSet<String>>();
    let digits = |w: &BTreeSet<String>| w.iter().filter(|t| t.chars().any(|c| c.is_ascii_digit())).cloned().collect::<Vec<_>>();
    let (wa, wb) = (words(a), words(b));
    if wa.is_empty() || wb.is_empty() || digits(&wa) != digits(&wb) {
        return false;
    }
    if a.contains(b) || b.contains(a) {
        return true;
    }
    let shared = wa.intersection(&wb).count();
    let all = wa.union(&wb).count();
    shared * 10 >= all * 7
}

/// The text states this date: its day as a number, its month by name and, when the date was not
/// inferred, its year ("DOF publicado en 18 enero 2024" states 2024-01-18).
fn text_states_date(text: &str, d: &DateItem) -> bool {
    const MONTHS: [&str; 12] = ["enero", "febrero", "marzo", "abril", "mayo", "junio", "julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre"];
    let part = |a: usize, z: usize| d.start.get(a..z).and_then(|t| t.parse::<usize>().ok());
    let (Some(year), Some(month), Some(day)) = (part(0, 4), part(5, 7), part(8, 10)) else { return false };
    let lower = text.to_lowercase();
    let tokens: BTreeSet<&str> = lower.split(|c: char| !c.is_alphanumeric()).filter(|t| !t.is_empty()).collect();
    let month_name = MONTHS.get(month.wrapping_sub(1)).copied().unwrap_or("");
    tokens.contains(month_name) && tokens.iter().any(|t| t.trim_start_matches('0') == day.to_string()) && (d.year_inferred || tokens.contains(year.to_string().as_str()))
}

// ------------------------------------------------------------------ what the model sees

/// The candidates worth labeling. Repeated pages are not sent, nor are the plain lines of the pages
/// of an annex of documents that the rules read whole. (The bullets of a page titled "Rubros
/// Financiables" used to be left out as the rules' own: that was true for the Monte de Piedad and
/// removed the whole list of what is and is not funded from Alsea.) If what is left exceeds the budget,
/// the weakest candidates go first (the order of the document is kept). Returns the candidates
/// and how many were dropped for the budget.
pub fn label_set(x: &CallExtraction, candidates: &[Candidate], budget_chars: usize) -> (Vec<Candidate>, usize) {
    let score = |c: &Candidate| -> i32 {
        c.signals
            .iter()
            .map(|s| match *s {
                "obligation" | "exclusion" | "priority" | "participation" | "rule_section" | "table" => 3,
                "limit" | "date" => 2,
                _ => 1,
            })
            .max()
            .unwrap_or(0)
    };
    // the annex of documents is read whole by the rules: its description column, sent as candidates,
    // came back from one model as 5 extra documents that duplicate the numbered ones
    let annex_pages: BTreeSet<usize> = x.documents.iter().filter(|d| d.source == "annex").map(|d| d.evidence.page).collect();
    let mut kept: Vec<(usize, &Candidate)> = candidates
        .iter()
        .enumerate()
        .filter(|(_, c)| !x.duplicate_pages.iter().any(|(d, _)| *d == c.page))
        .filter(|(_, c)| !(annex_pages.contains(&c.page) && !c.signals.iter().any(|s| ["date", "limit", "exclusion", "priority"].contains(s))))
        .collect();
    let size = |c: &Candidate| c.render().chars().count() + 1;
    let mut total: usize = kept.iter().map(|(_, c)| size(c)).sum();
    let mut dropped = 0;
    if total > budget_chars {
        // weakest first; among equals the later ones, so the start of the document survives
        let mut order: Vec<usize> = (0..kept.len()).collect();
        order.sort_by(|&a, &b| score(kept[a].1).cmp(&score(kept[b].1)).then(kept[b].0.cmp(&kept[a].0)));
        let mut drop: BTreeSet<usize> = BTreeSet::new();
        for i in order {
            if total <= budget_chars {
                break;
            }
            total -= size(kept[i].1);
            drop.insert(i);
        }
        dropped = drop.len();
        kept = kept.into_iter().enumerate().filter(|(i, _)| !drop.contains(i)).map(|(_, k)| k).collect();
    }
    (kept.into_iter().map(|(_, c)| c.clone()).collect(), dropped)
}

fn context_of(sent: &[Candidate]) -> String {
    let lines: Vec<String> = sent.iter().map(Candidate::render).collect();
    format!("Fragmentos de la convocatoria ({}):\n{}", sent.len(), lines.join("\n"))
}

// ------------------------------------------------------------------ reading the answer

fn str_of<'a>(v: &'a Value, key: &str) -> &'a str {
    v[key].as_str().unwrap_or("").trim()
}

/// The ids a model answer labeled validly (a known label).
fn labeled_ids(answer: &Value) -> BTreeSet<String> {
    answer["labels"]
        .as_array()
        .into_iter()
        .flatten()
        .filter(|l| LABELS.contains(&str_of(l, "label")))
        .map(|l| str_of(l, "id").to_string())
        .collect()
}

type Labels = BTreeMap<String, (&'static str, &'static str)>;

/// The label of every candidate: the first valid answer for an id wins (the first call, then the
/// repeat), so the result does not depend on the order the model listed them.
fn collect_labels(answers: &[Value], sent: &[Candidate], report: &mut ReviewReport) -> Labels {
    let known: BTreeMap<&str, &Candidate> = sent.iter().map(|c| (c.id.as_str(), c)).collect();
    let mut out: Labels = BTreeMap::new();
    for answer in answers {
        for l in answer["labels"].as_array().into_iter().flatten() {
            let id = str_of(l, "id");
            let what = format!("etiqueta de «{id}»");
            let Some(candidate) = known.get(id) else {
                report.rejected.push(Rejected { what, why: "ese id no existe entre los fragmentos enviados".into() });
                continue;
            };
            let Some(label) = LABELS.iter().find(|x| **x == str_of(l, "label")) else {
                report.rejected.push(Rejected { what, why: "la etiqueta no está en la lista permitida".into() });
                continue;
            };
            // whom a candidate applies to is read by the code from its own words and its list's
            out.entry(id.to_string()).or_insert((*label, derive_applies(candidate)));
        }
    }
    out
}

/// Whom a candidate applies to, read from its own words and the line that opened its list. The
/// model used to choose and two models agreed on it barely half the time ("for new organizations"
/// versus "for everyone" is a judgement call), and the code can read it: the words are fixed.
/// Order matters: a document for new organizations "of the category 2" is for new organizations.
fn derive_applies(c: &Candidate) -> &'static str {
    let hay = format!("{} {}", c.lead.as_deref().unwrap_or(""), c.text).to_lowercase();
    let has = |re: &Regex| re.is_match(&hay);
    if has(re!(r"nunca\s+han\s+recibido|organizaciones\s+nuevas|por\s+primera\s+vez|no\s+recibieron\s+donativ|no\s+cuentan\s+con\s+(un\s+)?registro")) {
        "new_organizations"
    } else if has(re!(r"categor[ií]a\s+\d|seg[uú]n\s+la\s+categor|en\s+el\s+caso\s+de\s+la\s+categor")) {
        "per_category"
    } else if has(re!(r"\betapa\b|si\s+tu\s+organizaci[oó]n\s+ha\s+aprobado")) {
        "per_stage"
    } else {
        "all"
    }
}

/// "(aplica a: …)" in the words of the closed list, never in the model's.
fn applies_phrase(applies: &str, lead: Option<&str>) -> String {
    match applies {
        "new_organizations" => "organizaciones nuevas".into(),
        "per_category" => "según la categoría".into(),
        "per_stage" => "según la etapa".into(),
        "other" => lead.map(|l| l.chars().take(80).collect()).unwrap_or_else(|| "un caso particular".into()),
        _ => String::new(),
    }
}

fn date_slot_kind(slot: &str) -> Option<&'static str> {
    ["application_window", "registration", "info_session", "disbursement"].into_iter().find(|k| *k == slot)
}

fn limit_slot_kind(slot: &str) -> Option<&'static str> {
    ["max_duration_months", "max_admin_percent", "min_operating_years", "max_amount", "min_amount"].into_iter().find(|k| *k == slot)
}

// ------------------------------------------------------------------ merging what was read

/// Adds the dates read from a candidate. `kind` is what the model says this date is (the fixed
/// question it answered); it settles a date the rules could only call "other", and when the rules
/// already gave another kind it is listed as a disagreement, not applied.
fn merge_dates(
    x: &mut CallExtraction,
    parsed: Vec<DateItem>,
    kind: Option<&'static str>,
    applied: &mut Vec<String>,
    added: &mut Vec<(String, String)>,
    disagreements: &mut Vec<String>,
    read: &mut Vec<String>,
) {
    // A candidate with several dates (a stage list, a paragraph with two deadlines) cannot tell
    // which one the question is about: their kinds stay as the rules read them, and it is said.
    let settles = parsed.len() == 1;
    if kind.is_some() && !settles && !parsed.is_empty() {
        disagreements.push(format!("el modelo señala un fragmento con {} fechas para {}: no se sabe cuál es, se dejan como las leyeron las reglas", parsed.len(), kind.unwrap_or("")));
    }
    for mut d in parsed {
        if let Some(k) = kind.filter(|_| settles) {
            d.kind = k;
        }
        // "Tienes hasta el 23 de mayo": the last day of the window 16–23 mayo, not one more date
        if d.end.is_none() && x.dates.iter().any(|e| e.kind == d.kind && e.end.as_deref() == Some(d.start.as_str())) {
            continue;
        }
        read.push(match &d.end {
            Some(e) => format!("{} → {e}", d.start),
            None => d.start.clone(),
        });
        if let Some(existing) = x.dates.iter_mut().find(|e| e.start == d.start && e.end == d.end) {
            if existing.kind == d.kind || d.kind == "other" {
                continue;
            }
            if existing.kind == "other" {
                applied.push(format!("clasificó la fecha {} como {}", d.start, d.kind));
                existing.kind = d.kind;
            } else {
                disagreements.push(format!("fecha {}: las reglas dicen «{}» y el modelo «{}»", d.start, existing.kind, d.kind));
            }
            continue;
        }
        applied.push(format!("agregó fecha {} {}", d.kind, d.start));
        added.push((d.evidence.snippet.clone(), String::new()));
        x.dates.push(d);
    }
}

/// Adds the limits read from a candidate: with `only`, just that kind. A value already there is
/// skipped; a different one goes beside it and the limit becomes conditional.
fn merge_limits(
    x: &mut CallExtraction,
    parsed: Vec<Limit>,
    only: Option<&str>,
    applied: &mut Vec<String>,
    added: &mut Vec<(String, String)>,
    read: &mut Vec<String>,
) {
    for l in parsed.into_iter().filter(|l| only.map_or(true, |k| l.kind == k)) {
        read.push(format!("{} = {}", l.kind, l.value));
        if let Some(existing) = x.limits.iter_mut().find(|e| e.kind == l.kind) {
            if (existing.value - l.value).abs() < f64::EPSILON || existing.others.contains(&l.value) {
                continue;
            }
            existing.others.push(l.value);
            existing.conditional = true;
            applied.push(format!("otro valor para {}: {}", l.kind, l.value));
        } else {
            applied.push(format!("agregó límite {} = {}", l.kind, l.value));
            added.push((l.evidence.snippet.clone(), String::new()));
            x.limits.push(l);
        }
    }
}

// ------------------------------------------------------------------ applying the answer

/// Merges the model's answers into the rules' extraction. Pure and deterministic: the same
/// extraction, candidates and answers always give the same draft, whatever order the model listed
/// its labels in. `answers` is the first reply and, if the model skipped ids, the repeat.
pub fn apply_labels(mut x: CallExtraction, sent: &[Candidate], dropped: usize, answers: &[Value]) -> CallDraft {
    let mut report = ReviewReport::without_ai("used");
    report.candidates_sent = sent.len();
    report.candidates_dropped = dropped;
    report.pages_sent = sent.iter().map(|c| c.page).collect::<BTreeSet<_>>().into_iter().collect();
    report.chars_sent = sent.iter().map(|c| c.text.chars().count()).sum();
    report.asked_again = answers.len() > 1;

    let labels = collect_labels(answers, sent, &mut report);
    report.labeled = labels.len();
    report.unlabeled = sent.iter().filter(|c| !labels.contains_key(&c.id)).map(|c| c.id.clone()).collect();
    for (label, _) in labels.values() {
        *report.by_label.entry((*label).to_string()).or_default() += 1;
    }

    let mut profile = CallProfile::default();
    // (snippet, applies phrase) of everything the model added: the draft that comes from it is marked
    let mut added: Vec<(String, String)> = Vec::new();
    let mut known_criteria: Vec<String> = x.eligibility.iter().map(|c| norm(&c.text)).collect();
    let mut known_docs: Vec<String> = x.documents.iter().map(|d| norm(&d.text)).collect();
    let year = x.year;

    // 1. each candidate, in the order of the document
    for c in sent {
        let Some(&(label, applies)) = labels.get(&c.id) else { continue };
        report.labels_by_id.insert(c.id.clone(), (label.to_string(), applies.to_string()));
        let phrase = applies_phrase(applies, c.lead.as_deref());
        let evidence = Evidence { page: c.page, snippet: c.text.clone() };
        let item = |label: &'static str| ProfileItem { text: c.text.clone(), label, applies_to: applies, evidence: evidence.clone(), candidate: c.id.clone() };
        let mut as_criterion = |x: &mut CallExtraction, report: &mut ReviewReport| {
            let n = norm(&c.text);
            if known_criteria.iter().any(|k| same_statement(k, &n)) {
                return;
            }
            known_criteria.push(n);
            x.eligibility.push(TextItem { text: c.text.clone(), evidence: evidence.clone() });
            report.applied.push(format!("agregó criterio: {}", c.text.chars().take(80).collect::<String>()));
            added.push((c.text.clone(), phrase.clone()));
        };
        match label {
            "criterion" => as_criterion(&mut x, &mut report),
            "exclusion" => profile.restrictions.push(item("exclusion")),
            "funded" => profile.funded.push(item("funded")),
            "population" => profile.population.push(item("population")),
            "calendar" => profile.calendar.push(item("calendar")),
            "submission" => profile.submission.push(item("submission")),
            "project_requirement" => profile.project_requirements.push(item("project_requirement")),
            "funding_condition" => profile.funding_conditions.push(item("funding_condition")),
            "priority" => profile.evaluation.push(item("priority")),
            "evaluation" => profile.evaluation.push(item("evaluation")),
            "document" => {
                let n = norm(&c.text);
                if !known_docs.iter().any(|k| same_statement(k, &n)) {
                    known_docs.push(n);
                    x.documents.push(DocumentItem {
                        number: None,
                        text: c.text.clone(),
                        clean_name: c.text.chars().count() <= 90,
                        source: "review",
                        evidence: evidence.clone(),
                    });
                    report.applied.push(format!("agregó documento: {}", c.text.chars().take(80).collect::<String>()));
                    added.push((c.text.clone(), phrase.clone()));
                }
            }
            "objective" if profile.objective.is_none() => profile.objective = Some(item("objective")),
            "funder" if profile.funder.is_none() => profile.funder = Some(item("funder")),
            "category" => {
                if !profile.agendas.iter().any(|a| norm(&a.name) == norm(&c.text)) {
                    profile.agendas.push(AgendaItem { name: c.text.clone(), evidence: evidence.clone(), candidate: c.id.clone() });
                }
            }
            _ => {}
        }
    }

    // 2. the fixed questions: answered from the first reply (the repeat only asks about labels)
    let by_id: BTreeMap<&str, &Candidate> = sent.iter().map(|c| (c.id.as_str(), c)).collect();
    let first = answers.first().cloned().unwrap_or(Value::Null);
    for slot in SLOTS {
        let entry = first["fields"].as_array().into_iter().flatten().find(|f| str_of(f, "slot") == slot);
        let Some(entry) = entry else {
            report.fields.push(FieldReport { slot: slot.into(), status: "unanswered".into(), ids: vec![], read: vec![] });
            continue;
        };
        let ids: Vec<String> = entry["ids"].as_array().into_iter().flatten().filter_map(|i| i.as_str()).map(str::to_string).collect();
        if str_of(entry, "status") != "found" {
            let rules_have = date_slot_kind(slot).is_some_and(|k| x.dates.iter().any(|d| d.kind == k)) || limit_slot_kind(slot).is_some_and(|k| x.limits.iter().any(|l| l.kind == k));
            if rules_have {
                report.disagreements.push(format!("{slot}: las reglas encontraron un valor y el modelo dice que no aparece en el texto"));
            }
            report.fields.push(FieldReport { slot: slot.into(), status: "not_in_text".into(), ids: vec![], read: vec![] });
            continue;
        }
        let mut read = Vec::new();
        let mut valid = Vec::new();
        for id in &ids {
            match by_id.get(id.as_str()) {
                Some(c) => {
                    valid.push(id.clone());
                    if let Some(kind) = date_slot_kind(slot) {
                        merge_dates(&mut x, dates_in_loose(&c.text, c.page, year), Some(kind), &mut report.applied, &mut added, &mut report.disagreements, &mut read);
                    } else {
                        merge_limits(&mut x, limits_in(&c.text, c.page), Some(slot), &mut report.applied, &mut added, &mut read);
                    }
                }
                None => report.rejected.push(Rejected { what: format!("pregunta {slot}"), why: format!("el id «{id}» no existe entre los fragmentos enviados") }),
            }
        }
        let status = if read.is_empty() {
            report.disagreements.push(format!("{slot}: el modelo señala {ids:?}, pero no se pudo leer un valor de ahí"));
            "unreadable"
        } else {
            "found"
        };
        report.fields.push(FieldReport { slot: slot.into(), status: status.into(), ids: valid, read });
    }

    // 3. a lone date read out of a document line ("DOF publicado en 18 enero 2024") is not a date
    //    of the process: it goes, with the reason, when the model says that line is a document
    let documents: Vec<&Candidate> = sent.iter().filter(|c| labels.get(&c.id).is_some_and(|(l, _)| *l == "document")).collect();
    let mut removed: Vec<String> = Vec::new();
    x.dates.retain(|d| {
        let inside_a_document = d.kind == "other" && d.end.is_none() && documents.iter().any(|c| c.page == d.evidence.page && text_states_date(&c.text, d));
        if inside_a_document {
            removed.push(format!("quitó la fecha {}: está en una línea que es un documento", d.start));
        }
        !inside_a_document
    });
    report.applied.extend(removed);

    // 4. a criterion of the rules that the model calls nothing: listed for a person
    let mut doubtful: Vec<Review> = Vec::new();
    for crit in &x.eligibility {
        let n = norm(&crit.text);
        let Some((c, label)) = sent.iter().find_map(|c| {
            let (label, _) = labels.get(&c.id)?;
            (c.page == crit.evidence.page && *label == "not_relevant" && same_statement(&norm(&c.text), &n)).then_some((c, *label))
        }) else {
            continue;
        };
        report.disagreements.push(format!("criterio de las reglas p.{}: el modelo lo etiquetó como «{label}»: {}", c.page, c.text.chars().take(80).collect::<String>()));
        doubtful.push(Review { reason: format!("Las reglas lo toman como criterio, pero el modelo lo etiquetó como «{label}». Revísalo."), page: c.page, snippet: c.text.clone() });
    }
    x.needs_review.extend(doubtful);

    profile.doubts = first["doubts"]
        .as_array()
        .into_iter()
        .flatten()
        .filter_map(|d| d.as_str())
        .map(|d| d.trim().chars().take(300).collect::<String>())
        .filter(|d| !d.is_empty())
        .take(10)
        .collect();

    // 5. requirement drafts again, with what the model added marked as its assumption
    if matches!(x.doc_kind, DocKind::FundingCall | DocKind::Unknown) {
        x.requirements = requirement_drafts(&x);
        for r in &mut x.requirements {
            if let Some((_, phrase)) = added.iter().find(|(snippet, _)| *snippet == r.source.snippet) {
                // What the model added is an assumption until a person confirms it, so it never
                // blocks a project on its own; and a rule that applies only to a case says so
                // (a real run added the 2024 financial statements, which only new organizations give).
                r.origin = "ai_assumption";
                r.blocking = false;
                if !phrase.is_empty() {
                    r.text = format!("{} (aplica a: {phrase})", r.text);
                }
            }
        }
    }
    CallDraft { extraction: x, profile, report }
}

// ------------------------------------------------------------------ agreement between two models

#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct LabelDiff {
    pub id: String,
    pub a: (String, String),
    pub b: (String, String),
}

/// How much two reviews of the SAME candidates agree: the measure of the rail. If a cheap model
/// and an expensive one label alike, the model can be swapped without the result changing.
#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct Agreement {
    /// Candidates both labeled.
    pub compared: usize,
    pub same_label: usize,
    /// Of those with the same label, how many also agree on whom it applies to.
    pub same_applies: usize,
    pub differing: Vec<LabelDiff>,
    /// Fragments that at least one model sent to criteria or documents (what the label of a fragment
    /// feeds in the product), and how many of them both sent to the same one. Disagreeing on whether
    /// a paragraph is the funder or the objective of the call changes nothing a project depends on.
    pub critical_compared: usize,
    pub critical_same: usize,
    pub fields_compared: usize,
    pub fields_same: usize,
    /// Fixed questions answered differently (status or what the code read).
    pub fields_differing: Vec<String>,
}

impl Agreement {
    pub fn label_rate(&self) -> f64 {
        if self.compared == 0 { 1.0 } else { self.same_label as f64 / self.compared as f64 }
    }

    pub fn critical_rate(&self) -> f64 {
        if self.critical_compared == 0 { 1.0 } else { self.critical_same as f64 / self.critical_compared as f64 }
    }
}

pub fn agreement(a: &ReviewReport, b: &ReviewReport) -> Agreement {
    let mut out = Agreement { compared: 0, same_label: 0, same_applies: 0, differing: vec![], critical_compared: 0, critical_same: 0, fields_compared: 0, fields_same: 0, fields_differing: vec![] };
    // what a label feeds in the product: criteria and documents. Dates and limits come from the rules
    // and the fixed questions (compared apart); the rest of the labels feeds the profile or nothing.
    let class = |label: &str| match label {
        "criterion" | "exclusion" => "criteria",
        "document" => "documents",
        _ => "other",
    };
    for (id, la) in &a.labels_by_id {
        let Some(lb) = b.labels_by_id.get(id) else { continue };
        out.compared += 1;
        let (ca, cb) = (class(&la.0), class(&lb.0));
        if ca != "other" || cb != "other" {
            out.critical_compared += 1;
            if ca == cb {
                out.critical_same += 1;
            }
        }
        if la.0 == lb.0 {
            out.same_label += 1;
            if la.1 == lb.1 {
                out.same_applies += 1;
            }
        } else {
            out.differing.push(LabelDiff { id: id.clone(), a: la.clone(), b: lb.clone() });
        }
    }
    for fa in &a.fields {
        let Some(fb) = b.fields.iter().find(|f| f.slot == fa.slot) else { continue };
        out.fields_compared += 1;
        let (ra, rb): (BTreeSet<&String>, BTreeSet<&String>) = (fa.read.iter().collect(), fb.read.iter().collect());
        if fa.status == fb.status && ra == rb {
            out.fields_same += 1;
        } else {
            out.fields_differing.push(format!("{}: {} {:?} contra {} {:?}", fa.slot, fa.status, fa.read, fb.status, fb.read));
        }
    }
    out
}

// ------------------------------------------------------------------ the review itself

async fn ask(
    provider: &dyn AiProvider,
    scanner: &dyn SensitiveScanner,
    ledger: &dyn Ledger,
    project_id: Option<String>,
    sent: &[Candidate],
) -> Result<Value, AiError> {
    let call = AiCall {
        task: AiTask::CallReview,
        context: vec![context_of(sent)],
        user: "Etiquete cada fragmento y responda las preguntas fijas.".into(),
        project_id,
    };
    pipeline::run(provider, scanner, ledger, call).await
}

/// Reads the call with the model, once, and merges the result. If the model skips some ids it is
/// asked once more about only those. Never fails: if the review cannot be done (no key, no
/// internet, no allowance) the draft is the rules' extraction and the report says why, so
/// uploading a call is never blocked by the AI.
pub async fn review_call(
    provider: &dyn AiProvider,
    scanner: &dyn SensitiveScanner,
    ledger: &dyn Ledger,
    project_id: Option<String>,
    x: CallExtraction,
    pages: &[PageText],
) -> CallDraft {
    let skip = |x: CallExtraction, ai: &str| CallDraft { extraction: x, profile: CallProfile::default(), report: ReviewReport::without_ai(ai) };
    // There is no gate that decides whether a document "is a call": the rules' guess made the rail
    // skip the whole Bienestar rules (ADR-014). A document with nothing to label gets labels of
    // `not_relevant`, which costs a few cents.
    let (sent, dropped) = label_set(&x, &segment(pages), LABEL_BUDGET_CHARS);
    if sent.is_empty() {
        return skip(x, "not_needed");
    }
    let first = match ask(provider, scanner, ledger, project_id.clone(), &sent).await {
        Ok(v) => v,
        Err(e) => return skip(x, &e.kind()),
    };
    let mut answers = vec![first];
    let done = labeled_ids(&answers[0]);
    let missing: Vec<Candidate> = sent.iter().filter(|c| !done.contains(&c.id)).cloned().collect();
    if !missing.is_empty() {
        // only about what was skipped; if this fails too, those ids stay unlabeled and are listed
        if let Ok(second) = ask(provider, scanner, ledger, project_id, &missing).await {
            answers.push(second);
        }
    }
    apply_labels(x, &sent, dropped, &answers)
}

    fn report_with(labels: &[(&str, &str, &str)], fields: &[(&str, &str, &[&str])]) -> ReviewReport {
        let mut r = ReviewReport::without_ai("used");
        for (id, label, applies) in labels {
            r.labels_by_id.insert((*id).into(), ((*label).into(), (*applies).into()));
        }
        for (slot, status, read) in fields {
            r.fields.push(FieldReport { slot: (*slot).into(), status: (*status).into(), ids: vec![], read: read.iter().map(|s| s.to_string()).collect() });
        }
        r
    }

    #[test]
    fn two_reviews_that_say_the_same_agree_completely_and_a_difference_is_named() {
        let a = report_with(&[("p1-1", "criterion", "all"), ("p1-2", "document", "new_organizations"), ("p1-3", "priority", "all")], &[("info_session", "found", &["2024-04-10T11:00"]), ("max_amount", "not_in_text", &[])]);
        assert_eq!(agreement(&a, &a.clone()).label_rate(), 1.0);
        let same = agreement(&a, &a.clone());
        assert_eq!((same.compared, same.same_applies, same.fields_same), (3, 3, 2));
        assert_eq!((same.critical_compared, same.critical_same), (2, 2), "criterion and document feed the product, priority does not");
        assert!(same.differing.is_empty() && same.fields_differing.is_empty());

        let b = report_with(
            &[("p1-1", "criterion", "all"), ("p1-2", "document", "all"), ("p1-3", "not_relevant", "all"), ("p9-9", "funder", "all")],
            &[("info_session", "found", &["2024-05-23T11:00"]), ("max_amount", "not_in_text", &[])],
        );
        let d = agreement(&a, &b);
        assert_eq!(d.compared, 3, "a candidate only one of them labeled is not compared");
        assert_eq!((d.same_label, d.same_applies), (2, 1));
        // p1-3 priority against not_relevant is not a difference the product feels
        assert_eq!((d.critical_compared, d.critical_same), (2, 2));
        assert_eq!(d.differing.len(), 1);
        assert_eq!(d.differing[0].id, "p1-3");
        assert_eq!((d.fields_compared, d.fields_same), (2, 1));
        assert!(d.fields_differing[0].starts_with("info_session"));
        assert!((d.label_rate() - 2.0 / 3.0).abs() < 1e-9);
    }

        let schema = crate::ai::prompts::schema(AiTask::CallReview);
