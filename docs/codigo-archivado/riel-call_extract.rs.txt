//! Deterministic extraction of a funding call (convocatoria) into structured JSON.
//! Rules first: the AI is only for what no rule can settle, and a person confirms it
//! (docs/10-metodologia-conversacion.md §13, ADR-010).
//!
//! Every extracted fact carries the page and the sentence it came from, so a person can check
//! it. What the rules cannot settle (tables read as text, values that live outside the
//! document) is listed in `needs_review` and `external`, never guessed.
//!
//! Scope of this first version: PDF with a text layer, Spanish. Scanned PDFs are reported
//! (`needs_ocr`); forms and linked attachments are listed (`links`) but not fetched.

// Not wired to a command yet (Phase 3): tests and the measurement run use it.

macro_rules! re {
    ($p:expr) => {{
        static R: OnceLock<Regex> = OnceLock::new();
        R.get_or_init(|| Regex::new($p).expect("valid regex"))
    }};
}

const MONTHS: [(&str, u32); 13] = [
    ("enero", 1), ("febrero", 2), ("marzo", 3), ("abril", 4), ("mayo", 5), ("junio", 6), ("julio", 7),
    ("agosto", 8), ("septiembre", 9), ("setiembre", 9), ("octubre", 10), ("noviembre", 11), ("diciembre", 12),
];
const MONTH_ALT: &str = "enero|febrero|marzo|abril|mayo|junio|julio|agosto|septiembre|setiembre|octubre|noviembre|diciembre";

// ------------------------------------------------------------------ reading text

/// Reads pages written as `=== PAGINA n ... ===` blocks (the format the dump tests write).
pub fn parse_dump(dump: &str) -> Vec<PageText> {
    let marker = re!(r"(?m)^=== PAGINA (\d+)[^\n]*===[ \t]*$");
    let marks: Vec<(usize, usize)> = marker
        .captures_iter(dump)
        .map(|c| (c.get(0).unwrap().end(), c[1].parse().unwrap_or(0)))
        .collect();
    let starts: Vec<usize> = marker.find_iter(dump).map(|m| m.start()).collect();
    marks
        .iter()
        .enumerate()
        .map(|(i, (end, number))| {
            let next = starts.get(i + 1).copied().unwrap_or(dump.len());
            PageText { number: *number, text: dump[*end..next].trim_matches('\n').to_string() }
        })
        .collect()
}

/// Repairs the letters some PDF fonts decode wrongly. In Spanish text `æ`, `Æ`, `Ø`, `œ` and
/// `˝` never appear, so when they sit inside words they are the accented letters of a font with
/// a custom encoding: `aæo` is `año`, `rØgimen` is `régimen`.
fn repair_mojibake(s: &str) -> Option<String> {
    if !re!(r"\p{L}[æÆØœ]\p{L}|˝\p{L}").is_match(s) {
        return None;
    }
    Some(
        s.chars()
            .map(|c| match c {
                'æ' => 'ñ',
                'Æ' => 'á',
                'Ø' => 'é',
                'œ' => 'ú',
                '˝' => 'Í',
                other => other,
            })
            .collect(),
    )
}

/// Normalizes text for matching: ligatures, odd spaces, broken URLs, wrongly decoded letters.
/// Returns the text and whether letters were repaired.
pub fn clean_text_report(raw: &str) -> (String, bool) {
    let raw = raw.replace("\r\n", "\n");
    let mut s = String::with_capacity(raw.len());
    for ch in raw.chars() {
        match ch {
            '\u{FB00}' => s.push_str("ff"),
            '\u{FB01}' => s.push_str("fi"),
            '\u{FB02}' => s.push_str("fl"),
            '\u{FB03}' => s.push_str("ffi"),
            '\u{FB04}' => s.push_str("ffl"),
            '\u{FB05}' | '\u{FB06}' => s.push_str("st"),
            '\u{00A0}' | '\u{2007}' | '\u{202F}' => s.push(' '),
            '\u{00AD}' | '\u{200B}' | '\u{FEFF}' => {}
            c => s.push(c),
        }
    }
    // some extractors leave the ligature as a literal "/f_i"
    let s = s.replace("/f_i", "fi").replace("/f_l", "fl").replace("/f_f", "ff");
    let (s, repaired) = match repair_mojibake(&s) {
        Some(fixed) => (fixed, true),
        None => (s, false),
    };
    let s = s.replace("https:/ /", "https://").replace("http:/ /", "http://");
    let s = re!(r"[ \t]+").replace_all(&s, " ").to_string();
    let s = re!(r" *\n *").replace_all(&s, "\n").to_string();
    let s = re!(r"\n{3,}").replace_all(&s, "\n\n").to_string();
    (s.trim().to_string(), repaired)
}

pub fn clean_text(raw: &str) -> String {
    clean_text_report(raw).0
}

/// A line that is a URL cut by the page width continues on the next line.
fn join_wrapped_urls(text: &str) -> String {
    let lines: Vec<&str> = text.lines().collect();
    let mut out: Vec<String> = Vec::new();
    let mut i = 0;
    while i < lines.len() {
        let mut line = lines[i].to_string();
        if line.contains("http") {
            while i + 1 < lines.len() {
                let next = lines[i + 1].trim();
                let url_like = !next.is_empty()
                    && !next.contains(' ')
                    && next.chars().any(|c| c.is_ascii_digit())
                    && next.chars().any(|c| c == '-' || c == '/' || c == '=')
                    && re!(r"^[A-Za-z0-9\-._~:/?#\[\]@!$&'()*+,;=%]+$").is_match(next);
                let last_token_is_url = line.split_whitespace().last().is_some_and(|t| t.starts_with("http"));
                if url_like && last_token_is_url {
                    line.push_str(next);
                    i += 1;
                } else {
                    break;
                }
            }
        }
        out.push(line);
        i += 1;
    }
    out.join("\n")
}

fn flat(s: &str) -> String {
    re!(r"\s+").replace_all(s, " ").trim().to_string()
}

/// The sentence-sized context around a match, on character boundaries.
fn clip(flat: &str, start: usize, end: usize) -> String {
    let mut a = start.saturating_sub(70);
    while a > 0 && !flat.is_char_boundary(a) {
        a -= 1;
    }
    let mut b = (end + 70).min(flat.len());
    while b < flat.len() && !flat.is_char_boundary(b) {
        b += 1;
    }
    flat[a..b].trim().to_string()
}

fn strip_accents(s: &str) -> String {
    s.chars()
        .map(|c| match c {
            'á' | 'Á' => 'a',
            'é' | 'É' => 'e',
            'í' | 'Í' => 'i',
            'ó' | 'Ó' => 'o',
            'ú' | 'Ú' | 'ü' => 'u',
            other => other.to_ascii_lowercase(),
        })
        .collect()
}

/// "tres" -> 3, "12" -> 12.
fn number_word(w: &str) -> Option<f64> {
    if let Ok(n) = w.replace(',', ".").parse::<f64>() {
        return Some(n);
    }
    const WORDS: [&str; 21] = [
        "cero", "uno", "dos", "tres", "cuatro", "cinco", "seis", "siete", "ocho", "nueve", "diez", "once", "doce", "trece",
        "catorce", "quince", "dieciseis", "diecisiete", "dieciocho", "diecinueve", "veinte",
    ];
    let w = strip_accents(w);
    WORDS.iter().position(|x| *x == w).map(|i| i as f64)
}

fn month_number(name: &str) -> Option<u32> {
    let n = name.to_lowercase();
    MONTHS.iter().find(|(m, _)| *m == n).map(|(_, v)| *v)
}

#[derive(Clone)]
struct FlatPage {
    number: usize,
    /// Cleaned text with its line breaks (for lists and tables).
    clean: String,
    /// The same text as one line (for sentences that wrap).
    flat: String,
}

// ------------------------------------------------------------------ limits

/// The text just before a match, on character boundaries.
fn before(flat: &str, start: usize, chars: usize) -> &str {
    let mut a = start.saturating_sub(chars);
    while a > 0 && !flat.is_char_boundary(a) {
        a -= 1;
    }
    &flat[a..start]
}

fn find_limits(pages: &[FlatPage]) -> Vec<Limit> {
    let mut out: Vec<Limit> = Vec::new();
    // A document can state the same limit more than once, with different values for different
    // categories. The first is kept as `value`; the rest go to `others`, and a value tied to a
    // category or a case is marked `conditional`, so it is never presented as the only one.
    let mut push = |kind: &'static str, value: f64, page: &FlatPage, start: usize, end: usize| {
        // only the sentence the value is in: a period after a lowercase letter ends it, but the
        // period of "Categoría 2. Educación" (after a number) does not
        let window = before(&page.flat, start, 140);
        let window = match re!(r"[\p{Ll}%)]\.\s").find_iter(window).last() {
            Some(m) => &window[m.end()..],
            None => window,
        };
        let conditional = re!(r"(?i)categor[ií]a|en\s+el\s+caso\s+de|seg[uú]n\s+el\s+tipo|para\s+los\s+proyectos\s+de")
            .is_match(&format!("{window}{}", &page.flat[start..end]));
        if let Some(l) = out.iter_mut().find(|l| l.kind == kind) {
            if (l.value - value).abs() > f64::EPSILON && !l.others.contains(&value) {
                l.others.push(value);
            }
            l.conditional |= conditional;
        } else {
            out.push(Limit { kind, value, others: Vec::new(), conditional, evidence: Evidence { page: page.number, snippet: clip(&page.flat, start, end) } });
        }
    };
    for p in pages {
        // "Máximo 12 meses", "duración máxima de hasta 12 meses", "hasta 12 meses"
        for c in re!(r"(?i)(?:m[aá]xim[oa]|hasta)\s+(?:de\s+)?(?:hasta\s+)?(\d{1,3})\s+meses").captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            push("max_duration_months", c[1].parse().unwrap_or(0.0), p, m.start(), m.end());
        }
        let admin = re!(
            r"(?i)gastos\s+administrativos[^.]{0,260}?(?:no\s+(?:podr[aá]n|deber[aá]n|pueden)\s+(?:superar|exceder|rebasar)|m[aá]ximo(?:\s+de)?|hasta)\s+(?:el\s+)?(\d{1,2}(?:[.,]\d+)?)\s*%"
        );
        if let Some(c) = admin.captures(&p.flat) {
            let m = c.get(0).unwrap();
            push("max_admin_percent", number_word(&c[1]).unwrap_or(0.0), p, m.start(), m.end());
        }
        let years = re!(
            r"(?i)al\s+menos\s+(\d{1,2}|[a-záéíóú]+)\s+a[nñ]os\s+de\s+(?:operaci[oó]n|experiencia|existencia|constituci[oó]n)"
        );
        if let Some(c) = years.captures(&p.flat) {
            if let Some(v) = number_word(&c[1]) {
                let m = c.get(0).unwrap();
                push("min_operating_years", v, p, m.start(), m.end());
            }
        }
        // amounts: "monto máximo de $500,000", "hasta $200,000"
        let amount = re!(r"(?i)(m[aá]ximo|m[ií]nimo|hasta)[^$.]{0,40}?\$\s?(\d[\d,.]*\d)");
        for c in amount.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            let digits = c[2].replace(',', "");
            let Ok(v) = digits.parse::<f64>() else { continue };
            let kind = if c[1].eq_ignore_ascii_case("mínimo") || c[1].eq_ignore_ascii_case("minimo") { "min_amount" } else { "max_amount" };
            push(kind, v, p, m.start(), m.end());
        }
    }
    out
}

fn find_legal_figures(pages: &[FlatPage]) -> Option<LegalFigures> {
    let re = re!(r"(?i)sin\s+fines\s+de\s+lucro\s*\(([A-Za-z.,\s]+)\)");
    for p in pages {
        if let Some(c) = re.captures(&p.flat) {
            // legal figures are written as capital acronyms (AC, IAP); a word such as "otras" is not one
            let values: Vec<String> = c[1]
                .split(',')
                .map(|t| t.replace('.', "").trim().to_string())
                .filter(|t| (2..=5).contains(&t.chars().count()) && t.chars().all(|ch| ch.is_uppercase()))
                .collect();
            if !values.is_empty() {
                let m = c.get(0).unwrap();
                return Some(LegalFigures { values, evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) } });
            }
        }
    }
    None
}

// ------------------------------------------------------------------ dates

fn nearest_kind(flat: &str, before_pos: usize) -> &'static str {
    let window = before(flat, before_pos, 320).to_lowercase();
    let groups: [(&'static str, &[&str]); 4] = [
        ("application_window", &["formulario", "postulaci"]),
        ("info_session", &["sesi"]),
        ("disbursement", &["entrega de recursos", "entrega del donativo", "entrega"]),
        ("registration", &["registro"]),
    ];
    groups
        .iter()
        .filter_map(|(kind, words)| words.iter().filter_map(|w| window.rfind(w)).max().map(|pos| (pos, *kind)))
        .max_by_key(|(pos, _)| *pos)
        .map(|(_, kind)| kind)
        .unwrap_or("other")
}

fn find_dates(pages: &[FlatPage], doc_year: Option<u32>) -> Vec<DateItem> {
    find_dates_with(pages, doc_year, false)
}

/// `loose` also reads a day and a month with no year and no time ("15 Julio"). On a whole page
/// that is too loose ("el 10 abril es feriado"), but on a line a model has already said holds a
/// date it is what the line says.
fn find_dates_with(pages: &[FlatPage], doc_year: Option<u32>, loose: bool) -> Vec<DateItem> {
    let mut out: Vec<DateItem> = Vec::new();
    for p in pages {
        let mut taken: Vec<(usize, usize)> = Vec::new();
        let free = |taken: &Vec<(usize, usize)>, s: usize, e: usize| !taken.iter().any(|(a, b)| s < *b && e > *a);

        let range = Regex::new(&format!(r"(?i)del\s+(\d{{1,2}})\s+al\s+(\d{{1,2}})\s+de\s+({MONTH_ALT})\s+(?:de\s+)?(\d{{4}})")).unwrap();
        for c in range.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            let (Some(mo), Ok(y), Ok(d1), Ok(d2)) = (month_number(&c[3]), c[4].parse::<u32>(), c[1].parse::<u32>(), c[2].parse::<u32>()) else { continue };
            taken.push((m.start(), m.end()));
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{mo:02}-{d1:02}"),
                end: Some(format!("{y:04}-{mo:02}-{d2:02}")),
                precision: "day",
                year_inferred: false,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }

        // Forms with no year or no "de": "19 MAYO-6 JUNIO", "15-24 SEPTIEMBRE", "A PARTIR DEL 23 DE
        // JUNIO". The same funder writes dates differently from one year to the next. When the text
        // gives no year, the document's own is used and the date is flagged.
        let cross = Regex::new(&format!(
            r"(?i)\b(\d{{1,2}})\s+(?:de\s+)?({MONTH_ALT})\s*[-–]\s*(\d{{1,2}})\s+(?:de\s+)?({MONTH_ALT})(?:\s+(?:de\s+)?(\d{{4}}))?"
        ))
        .unwrap();
        for c in cross.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            if !free(&taken, m.start(), m.end()) {
                continue;
            }
            let (Some(m1), Some(m2), Ok(d1), Ok(d2)) = (month_number(&c[2]), month_number(&c[4]), c[1].parse::<u32>(), c[3].parse::<u32>()) else { continue };
            let (y, inferred) = match (c.get(5).and_then(|y| y.as_str().parse::<u32>().ok()), doc_year) {
                (Some(y), _) => (y, false),
                (None, Some(y)) => (y, true),
                _ => continue,
            };
            taken.push((m.start(), m.end()));
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{m1:02}-{d1:02}"),
                end: Some(format!("{y:04}-{m2:02}-{d2:02}")),
                precision: "day",
                year_inferred: inferred,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }
        let same_month = Regex::new(&format!(r"(?i)\b(\d{{1,2}})\s*[-–]\s*(\d{{1,2}})\s+(?:de\s+)?({MONTH_ALT})(?:\s+(?:de\s+)?(\d{{4}}))?")).unwrap();
        for c in same_month.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            if !free(&taken, m.start(), m.end()) {
                continue;
            }
            let (Some(mo), Ok(d1), Ok(d2)) = (month_number(&c[3]), c[1].parse::<u32>(), c[2].parse::<u32>()) else { continue };
            let (y, inferred) = match (c.get(4).and_then(|y| y.as_str().parse::<u32>().ok()), doc_year) {
                (Some(y), _) => (y, false),
                (None, Some(y)) => (y, true),
                _ => continue,
            };
            taken.push((m.start(), m.end()));
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{mo:02}-{d1:02}"),
                end: Some(format!("{y:04}-{mo:02}-{d2:02}")),
                precision: "day",
                year_inferred: inferred,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }
        // a calendar written as the month and then the days: "Abril 01 al 22"
        let month_first = Regex::new(&format!(r"(?i)\b({MONTH_ALT})\s+(\d{{1,2}})\s+al\s+(\d{{1,2}})\b")).unwrap();
        for c in month_first.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            if !free(&taken, m.start(), m.end()) {
                continue;
            }
            let (Some(mo), Ok(d1), Ok(d2), Some(y)) = (month_number(&c[1]), c[2].parse::<u32>(), c[3].parse::<u32>(), doc_year) else { continue };
            taken.push((m.start(), m.end()));
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{mo:02}-{d1:02}"),
                end: Some(format!("{y:04}-{mo:02}-{d2:02}")),
                precision: "day",
                year_inferred: true,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }
        let from_day = Regex::new(&format!(r"(?i)a\s+partir\s+del?\s+(\d{{1,2}})\s+de\s+({MONTH_ALT})(?:\s+(?:de\s+)?(\d{{4}}))?")).unwrap();
        for c in from_day.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            if !free(&taken, m.start(), m.end()) {
                continue;
            }
            let (Some(mo), Ok(d)) = (month_number(&c[2]), c[1].parse::<u32>()) else { continue };
            let (y, inferred) = match (c.get(3).and_then(|y| y.as_str().parse::<u32>().ok()), doc_year) {
                (Some(y), _) => (y, false),
                (None, Some(y)) => (y, true),
                _ => continue,
            };
            taken.push((m.start(), m.end()));
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{mo:02}-{d:02}"),
                end: None,
                precision: "day",
                year_inferred: inferred,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }

        // A time follows as "12:30 p. m.", "11:00 hrs" or "a las 11:00 hrs": the 12-hour form carries
        // a/p, the others are 24-hour.
        let single = Regex::new(&format!(
            r"(?i)\b(\d{{1,2}})\s+(?:de\s+)?({MONTH_ALT})\s+(?:de\s+)?(\d{{4}})(?:,?\s+(?:a\s+las\s+)?(\d{{1,2}}):(\d{{2}})\s*(?:([ap])\s*\.?\s*m\.?|hrs?\.?|horas|h\b)?)?"
        ))
        .unwrap();
        for c in single.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            if !free(&taken, m.start(), m.end()) {
                continue;
            }
            let (Some(mo), Ok(y), Ok(d)) = (month_number(&c[2]), c[3].parse::<u32>(), c[1].parse::<u32>()) else { continue };
            taken.push((m.start(), m.end()));
            let (start, precision) = match (c.get(4), c.get(5), c.get(6)) {
                (Some(h), Some(min), Some(ap)) => {
                    let mut hour: u32 = h.as_str().parse().unwrap_or(0) % 12;
                    if ap.as_str().eq_ignore_ascii_case("p") {
                        hour += 12;
                    }
                    (format!("{y:04}-{mo:02}-{d:02}T{hour:02}:{}", min.as_str()), "datetime")
                }
                (Some(h), Some(min), None) if h.as_str().parse::<u32>().is_ok_and(|h| h < 24) && min.as_str().parse::<u32>().is_ok_and(|m| m < 60) => {
                    (format!("{y:04}-{mo:02}-{d:02}T{:02}:{}", h.as_str().parse::<u32>().unwrap_or(0), min.as_str()), "datetime")
                }
                _ => (format!("{y:04}-{mo:02}-{d:02}"), "day"),
            };
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start,
                end: None,
                precision,
                year_inferred: false,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }

        // "el 10 abril a las 11:00 hrs": a day and a month with a time but no year (a session). The
        // document's year is used and flagged. Without the time, "10 abril" alone is too loose to read.
        let no_year = Regex::new(&format!(r"(?i)\b(\d{{1,2}})\s+(?:de\s+)?({MONTH_ALT})\b,?\s+(?:a\s+las\s+)?(\d{{1,2}}):(\d{{2}})\s*(?:hrs?\.?|horas|h\b|([ap])\s*\.?\s*m\.?)")).unwrap();
        for c in no_year.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            if !free(&taken, m.start(), m.end()) {
                continue;
            }
            let (Some(mo), Some(y), Ok(d), Ok(h), Ok(min)) = (month_number(&c[2]), doc_year, c[1].parse::<u32>(), c[3].parse::<u32>(), c[4].parse::<u32>()) else { continue };
            if h > 23 || min > 59 {
                continue;
            }
            let hour = match c.get(5) {
                Some(ap) if ap.as_str().eq_ignore_ascii_case("p") => h % 12 + 12,
                Some(_) => h % 12,
                None => h,
            };
            taken.push((m.start(), m.end()));
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{mo:02}-{d:02}T{hour:02}:{min:02}"),
                end: None,
                precision: "datetime",
                year_inferred: true,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }

        if loose {
            let day_month = Regex::new(&format!(r"(?i)\b(\d{{1,2}})\s+(?:de\s+)?({MONTH_ALT})\b")).unwrap();
            for c in day_month.captures_iter(&p.flat) {
                let m = c.get(0).unwrap();
                if !free(&taken, m.start(), m.end()) {
                    continue;
                }
                let (Some(mo), Some(y), Ok(d)) = (month_number(&c[2]), doc_year, c[1].parse::<u32>()) else { continue };
                if !(1..=31).contains(&d) {
                    continue;
                }
                taken.push((m.start(), m.end()));
                out.push(DateItem {
                    kind: nearest_kind(&p.flat, m.start()),
                    start: format!("{y:04}-{mo:02}-{d:02}"),
                    end: None,
                    precision: "day",
                    year_inferred: true,
                    evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
                });
            }
        }

        // "A PARTIR DE JULIO": a month with no year; the document's year is used and flagged
        let from_month = Regex::new(&format!(r"(?i)a\s+partir\s+de\s+({MONTH_ALT})\b")).unwrap();
        for c in from_month.captures_iter(&p.flat) {
            let m = c.get(0).unwrap();
            let (Some(mo), Some(y)) = (month_number(&c[1]), doc_year) else { continue };
            out.push(DateItem {
                kind: nearest_kind(&p.flat, m.start()),
                start: format!("{y:04}-{mo:02}"),
                end: None,
                precision: "month",
                year_inferred: true,
                evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) },
            });
        }
    }
    // the same fact is often repeated on another page (a summary and the detail)
    let mut unique: Vec<DateItem> = Vec::new();
    for mut d in out {
        // A window to apply is a range. A lone date is only one when the cue is right around it:
        // "DOF publicado en 18 enero 2024" is a requirement about a document, not a window, even
        // when the word "formulario" appears a few lines above.
        if d.kind == "application_window" && d.end.is_none() {
            let around = d.evidence.snippet.to_lowercase();
            if !["formulario", "postulaci", "registro", "recepci", "fecha l"].iter().any(|w| around.contains(w)) {
                d.kind = "other";
            }
        }
        if !unique.iter().any(|u| u.kind == d.kind && u.start == d.start && u.end == d.end) {
            unique.push(d);
        }
    }
    unique
}

/// Where the application window stands on a given day (`YYYY-MM-DD`). Dates are compared as text,
/// which orders ISO dates correctly; no clock is read here so the extraction stays deterministic.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
#[serde(rename_all = "snake_case")]
pub enum WindowStatus {
    NotYet,
    Open,
    Closed,
    /// The document gives no application window.
    Unknown,
}

pub fn window_status(x: &CallExtraction, today: &str) -> WindowStatus {
    let Some(w) = x.dates.iter().find(|d| d.kind == "application_window" && d.precision == "day") else {
        return WindowStatus::Unknown;
    };
    let start = &w.start[..10.min(w.start.len())];
    let end = w.end.as_deref().map(|e| &e[..10.min(e.len())]).unwrap_or(start);
    if today < start {
        WindowStatus::NotYet
    } else if today > end {
        WindowStatus::Closed
    } else {
        WindowStatus::Open
    }
}

// ------------------------------------------------------------------ lists

fn is_heading(line: &str) -> bool {
    let letters: Vec<char> = line.chars().filter(|c| c.is_alphabetic()).collect();
    letters.len() >= 8 && letters.iter().all(|c| c.is_uppercase())
}

/// Bullets of the participation criteria and, apart, the documents listed under a bullet that
/// ends in ":". Several pages mention the section: the table of contents does (it has no
/// bullets) and a later page states more rules ("solo puedes participar en una sola agenda").
/// Criteria are gathered from every page that has bullets and mentions the section; the
/// document sub-list comes from the page that has it.
fn find_eligibility(pages: &[FlatPage], rule_pages: &[usize]) -> (Vec<TextItem>, Vec<DocumentItem>) {
    // A page counts when its TITLE is about participation (or it says "criterios mínimos", or it
    // is where the minimum years or the legal figures are stated), not when a sentence merely
    // mentions the words: page 9 of the real call does, and its bullets are procedure steps.
    // Bullets that start with a date or carry a link are sessions and steps.
    let about_participation = |p: &&FlatPage| {
        rule_pages.contains(&p.number)
            || re!(r"(?i)criterios\s+m[ií]nimos|criterios\s+a\s+considerar").is_match(&p.flat)
            || p.clean.lines().any(|l| {
                let t = l.trim();
                is_heading(t) && t.to_uppercase().contains("PARTICIPACI")
            })
    };
    let walked: Vec<(Vec<TextItem>, Vec<DocumentItem>)> = pages
        .iter()
        .filter(about_participation)
        .map(|p| split_by_blank_lines(&p.clean, p.number))
        .map(|(c, d)| (c.into_iter().filter(|i| !i.text.starts_with(|ch: char| ch.is_ascii_digit()) && !i.text.contains("http")).collect(), d))
        .filter(|(c, d): &(Vec<TextItem>, Vec<DocumentItem>)| !c.is_empty() || !d.is_empty())
        .collect();
    let docs = walked.iter().max_by_key(|(_, d)| d.len()).map(|(_, d)| d.clone()).unwrap_or_default();
    let mut criteria: Vec<TextItem> = Vec::new();
    for (items, _) in &walked {
        for item in items {
            if !criteria.iter().any(|c| c.text == item.text) {
                criteria.push(item.clone());
            }
        }
    }
    (criteria, docs)
}

/// Walks the bullets of a page: a bullet that ends in ":" opens a sub-list (the documents) and a
/// blank line closes it, which is the layout of the real calls.
fn split_by_blank_lines(clean: &str, page: usize) -> (Vec<TextItem>, Vec<DocumentItem>) {
    let mut criteria = Vec::new();
    let mut docs = Vec::new();
    let mut current: Option<String> = None;
    let mut sub = false;
    let mut opens = false;
    let flush = |cur: &mut Option<String>, sub: &mut bool, opens: &mut bool, criteria: &mut Vec<TextItem>, docs: &mut Vec<DocumentItem>| {
        if let Some(raw) = cur.take() {
            let text = flat(&raw);
            let ev = Evidence { page, snippet: text.clone() };
            if *sub {
                docs.push(DocumentItem { number: None, clean_name: text.chars().count() <= 90, source: "eligibility", evidence: ev, text });
            } else {
                *opens = text.ends_with(':');
                criteria.push(TextItem { text, evidence: ev });
                if *opens {
                    *sub = true;
                }
            }
        }
    };
    for line in clean.lines() {
        let t = line.trim();
        if t.is_empty() {
            flush(&mut current, &mut sub, &mut opens, &mut criteria, &mut docs);
            sub = false;
            continue;
        }
        if let Some(rest) = t.strip_prefix('•') {
            flush(&mut current, &mut sub, &mut opens, &mut criteria, &mut docs);
            current = Some(rest.trim().to_string());
            continue;
        }
        if is_heading(t) {
            flush(&mut current, &mut sub, &mut opens, &mut criteria, &mut docs);
            sub = false;
            continue;
        }
        if let Some(c) = current.as_mut() {
            // a bullet is a sentence: a line that starts a new sentence after a closing period is
            // the paragraph that follows the list, not a continuation of the last bullet
            let starts_new_sentence = c.trim_end().ends_with('.') && t.chars().next().is_some_and(char::is_uppercase);
            if starts_new_sentence {
                flush(&mut current, &mut sub, &mut opens, &mut criteria, &mut docs);
                sub = false;
            } else {
                c.push(' ');
                c.push_str(t);
            }
        }
    }
    flush(&mut current, &mut sub, &mut opens, &mut criteria, &mut docs);
    (criteria, docs)
}

/// The numbered document list of an annex ("11. Opinión de cumplimiento del SAT").
fn find_annex_documents(pages: &[FlatPage], review: &mut Vec<Review>) -> Vec<DocumentItem> {
    let mut out: Vec<DocumentItem> = Vec::new();
    for p in pages {
        // the matrix itself, not the table of contents (which also lists "ANEXO 1 GUÍA DE
        // DOCUMENTOS" and numbers the stages "1. Postulación", "2. Registro"...)
        let is_matrix = re!(r"(?i)nombre\s+del\s+documento|matriz\s+de\s+documentos\s+para").is_match(&p.flat);
        if !is_matrix || re!(r"(?i)\bcontenido\b").is_match(&p.flat) {
            continue;
        }
        for line in p.clean.lines() {
            let Some(c) = re!(r"^\s*(\d{1,2})\.\s+(\S.*)$").captures(line) else { continue };
            let number: u32 = c[1].parse().unwrap_or(0);
            if out.iter().any(|d| d.number == Some(number)) {
                continue;
            }
            let text = flat(&c[2]);
            out.push(DocumentItem {
                number: Some(number),
                clean_name: text.chars().count() <= 70,
                source: "annex",
                evidence: Evidence { page: p.number, snippet: flat(line) },
                text,
            });
        }
        // the stage columns come out as a column of "Si" with no row to attach them to
        let marks = p.clean.lines().filter(|l| l.trim().eq_ignore_ascii_case("si") || l.trim().eq_ignore_ascii_case("sí")).count();
        if marks >= 4 {
            review.push(Review {
                reason: "Matriz de documentos por etapa: las marcas «Sí» salen sin la fila a la que pertenecen; la etapa de cada documento no se puede asignar con el texto".into(),
                page: p.number,
                snippet: format!("{marks} marcas «Sí» sueltas"),
            });
        }
    }
    if let Some(max) = out.iter().filter_map(|d| d.number).max() {
        let missing: Vec<u32> = (1..=max).filter(|n| !out.iter().any(|d| d.number == Some(*n))).collect();
        if !missing.is_empty() {
            review.push(Review {
                reason: format!("En la lista numerada de documentos faltan los números {missing:?}"),
                page: out.first().map(|d| d.evidence.page).unwrap_or(0),
                snippet: String::new(),
            });
        }
    }
    out.sort_by_key(|d| d.number);
    out
}

fn concept_group(rubro: &str) -> &'static str {
    let r = strip_accents(rubro);
    if r.contains("inversi") {
        "inversion"
    } else if r.contains("operativ") {
        "operativo"
    } else if r.contains("administrativ") {
        "administrativo"
    } else {
        "otro"
    }
}

fn find_concepts(pages: &[FlatPage]) -> Vec<Concept> {
    let mut out = Vec::new();
    for p in pages {
        if !p.clean.contains("RUBRO:") {
            continue;
        }
        let mut group: Option<&'static str> = None;
        for line in p.clean.lines() {
            let t = line.trim();
            if let Some(rest) = t.strip_prefix("RUBRO:") {
                group = Some(concept_group(rest));
                continue;
            }
            let Some(g) = group else { continue };
            let skip = t.is_empty()
                || t.starts_with("NOTA")
                || t == "CONCEPTO"
                || t.chars().all(|c| c.is_ascii_digit())
                || t.to_uppercase().contains("LISTA DE RUBROS");
            if !skip {
                out.push(Concept { group: g, text: t.to_string(), page: p.number });
            }
        }
    }
    out
}

// ------------------------------------------------------------------ links, indicators, external

fn link_kind(url: &str) -> &'static str {
    let u = url.to_lowercase();
    if re!(r"\.(zip|pdf|xlsx?|docx?|rar)(\?|$)").is_match(&u) || u.contains("dl=1") || u.contains("download") {
        "download"
    } else if u.contains("tfaforms") || u.contains("forms.") || u.contains("/forms") || u.contains("formulario") || u.contains("/f/") {
        "form"
    } else if u.contains("events.") || u.contains("teams.") || u.contains("zoom.") || u.contains("meet.") || u.contains("/sesion") {
        "session"
    } else if u.contains("login") || u.contains("plataforma") {
        "platform"
    } else if u.contains("wa.me") || u.contains("whatsapp") {
        "contact"
    } else {
        "other"
    }
}

fn find_links(pages: &[FlatPage]) -> (Vec<Link>, Vec<String>) {
    let mut links: Vec<Link> = Vec::new();
    let mut emails: Vec<String> = Vec::new();
    for p in pages {
        let joined = join_wrapped_urls(&p.clean);
        for m in re!(r#"https?://[^\s<>"')]+"#).find_iter(&joined) {
            let url = m.as_str().trim_end_matches(['.', ',', ';', ':']).to_string();
            if !links.iter().any(|l| l.url == url) {
                links.push(Link { kind: link_kind(&url), url, page: p.number });
            }
        }
        // addresses written without the scheme ("www.sitio.com/ruta")
        for m in re!(r#"\bwww\.[\w-]+(?:\.[\w-]+)+(?:/[^\s<>"')]*)?"#).find_iter(&joined) {
            let bare = m.as_str().trim_end_matches(['.', ',', ';', ':']);
            if !links.iter().any(|l| l.url.ends_with(bare)) {
                let url = format!("https://{bare}");
                links.push(Link { kind: link_kind(&url), url, page: p.number });
            }
        }
        let without_urls = re!(r#"https?://[^\s<>"')]+"#).replace_all(&joined, " ");
        for m in re!(r"[\w.+-]+@[\w-]+(?:\.[\w-]+)+").find_iter(&without_urls) {
            let e = m.as_str().trim_end_matches('.').to_lowercase();
            if !emails.contains(&e) {
                emails.push(e);
            }
        }
    }
    (links, emails)
}

fn find_indicators(pages: &[FlatPage], review: &mut Vec<Review>) -> Vec<IndicatorItem> {
    let mut out = Vec::new();
    for p in pages {
        let has_table_header = re!(r"DIMENSI[OÓ]N\s+INDICADOR\s+DE\s+PRODUCTO").is_match(&p.flat);
        let is_indicator_page = has_table_header || re!(r"INDICADORES\s+PARA\s+PROYECTOS").is_match(&p.flat);
        if !is_indicator_page {
            continue;
        }
        let agenda = if re!(r"(?i)TRABAJO\s+Y\s+EMPRENDIMIENTO").is_match(&p.flat) && !re!(r"(?i)ACCESO\s*A\s*DERECHOS").is_match(&p.flat) {
            Some("trabajo_emprendimiento")
        } else if re!(r"(?i)ACCESO\s*A\s*DERECHOS|SERVICIOS\s+ASISTENCIALES").is_match(&p.flat) {
            Some("acceso_derechos")
        } else {
            None
        };
        let start = re!(r"^\s*(?:Porcentaje|Promedio|#)\s");
        let mut current: Option<String> = None;
        let mut flush = |cur: &mut Option<String>| {
            if let Some(t) = cur.take() {
                let text = flat(&t);
                if text.chars().count() > 12 {
                    out.push(IndicatorItem { text, agenda, page: p.number });
                }
            }
        };
        for line in p.clean.lines() {
            let t = line.trim();
            if t.is_empty() || is_heading(t) || t.starts_with("DIMENSIÓN") {
                flush(&mut current);
                continue;
            }
            if start.is_match(t) {
                flush(&mut current);
                current = Some(t.to_string());
            } else if let Some(c) = current.as_mut() {
                c.push(' ');
                c.push_str(t);
            }
        }
        flush(&mut current);
        if has_table_header {
            review.push(Review {
                reason: "Tabla de indicadores: con el texto no se puede asignar a cada indicador su dimensión ni si es de producto o de resultado".into(),
                page: p.number,
                snippet: "DIMENSIÓN · INDICADOR DE PRODUCTO O INSUMO · INDICADOR DE RESULTADO".into(),
            });
        }
    }
    out
}

fn find_external(pages: &[FlatPage]) -> Vec<External> {
    let mut out = Vec::new();
    let monto = re!(r"(?i)el\s+monto[^.]{0,140}?(?:se\s+encontrar[aá]|encontrar[aá]|podr[aá]s?\s+consultar|se\s+indicar[aá])[^.]{0,100}?plataforma");
    for p in pages {
        if let Some(m) = monto.find(&p.flat) {
            out.push(External { name: "monto", lives_in: "plataforma", evidence: Evidence { page: p.number, snippet: clip(&p.flat, m.start(), m.end()) } });
            break;
        }
    }
    out
}

// ------------------------------------------------------------------ tags, quality

fn tag_pages(pages: &[FlatPage]) -> Vec<PageTags> {
    let rules: [(&'static str, &Regex); 11] = [
        ("elegibilidad", re!(r"(?i)criterios\s+m[ií]nimos|criterios\s+a\s+considerar|requisitos\s+de\s+participaci[oó]n|criterios\s+obligatorios|podr[aá]n\s+participar")),
        ("fechas", re!(r"(?i)etapas\s+de\s+participaci[oó]n|calendario|fecha\s+l[ií]mite|\bdel\s+\d{1,2}\s+al\s+\d{1,2}\s+de\b")),
        ("documentos", re!(r"(?i)matriz\s+de\s+documentos|gu[ií]a\s+de\s+documentos")),
        ("financiable", re!(r"(?i)rubros?\s+financiables|gastos\s+de\s+inversi[oó]n|gastos\s+operativ|gastos\s+administrativ|lista\s+de\s+rubros")),
        ("indicadores", re!(r"(?i)indicadores\s+para\s+proyectos|indicador\s+de\s+producto")),
        ("poblacion", re!(r"(?i)poblaci[oó]n\s+objetivo|\bedades\b|^poblaci[oó]n")),
        ("evaluacion", re!(r"(?i)se\s+evaluar[aá]|criterios\s+de\s+evaluaci[oó]n|se\s+priorizar[aá]n|se\s+valorar[aá]")),
        ("estrategias", re!(r"(?i)estrategias?\s+de\b|modelos?\s+de\s+intervenci[oó]n")),
        ("contacto", re!(r"(?i)contacto|escr[ií]benos|queremos\s+acompa")),
        ("glosario", re!(r"(?i)glosario")),
        ("objetivo", re!(r"(?i)\bobjetivo\b")),
    ];
    pages
        .iter()
        .map(|p| {
            let mut tags: Vec<&'static str> = rules.iter().filter(|(_, r)| r.is_match(&p.flat)).map(|(t, _)| *t).collect();
            if re!(r"https?://").is_match(&p.flat) {
                tags.push("enlaces");
            }
            PageTags { page: p.number, tags }
        })
        .collect()
}

/// Pages where more than 2 % of the characters look like decoding noise.
fn is_low_quality(clean: &str) -> bool {
    let total = clean.chars().filter(|c| !c.is_whitespace()).count();
    if total < 20 {
        return false;
    }
    let bad = clean
        .chars()
        .filter(|c| {
            !(c.is_alphanumeric() || c.is_whitespace() || c.is_ascii_punctuation() || "¿¡•·…“”‘’–—°º€ª«»".contains(*c))
        })
        .count();
    bad * 50 > total
}

// ------------------------------------------------------------------ drafts for the requirement table

pub(crate) fn requirement_drafts(x: &CallExtraction) -> Vec<RequirementDraft> {
    let mut out: Vec<RequirementDraft> = Vec::new();
    let mut add = |text: String, kind: &'static str, value: serde_json::Value, blocking: bool, source: Evidence| {
        let code = format!("REQ-{:02}", out.len() + 1);
        out.push(RequirementDraft { code, text, kind, value, blocking, origin: "document", source });
    };
    if let Some(l) = &x.legal_figures {
        add(format!("Figura jurídica permitida: {}", l.values.join(", ")), "allowed_legal_figure", json!(l.values), true, l.evidence.clone());
    }
    for l in &x.limits {
        let (text, blocking) = match l.kind {
            "min_operating_years" => (format!("Operación continua mínima: {} años", l.value), true),
            "max_duration_months" => (format!("Duración máxima del proyecto: {} meses", l.value), true),
            "max_admin_percent" => (format!("Gastos administrativos: máximo {} % del donativo", l.value), true),
            "max_amount" => (format!("Monto máximo: ${}", l.value), true),
            "min_amount" => (format!("Monto mínimo: ${}", l.value), true),
            _ => continue,
        };
        // a value tied to a category, or with other values beside it, is not a rule for everyone:
        // it is offered to a person to confirm, never as a blocking requirement
        if l.conditional || !l.others.is_empty() {
            add(
                format!("{text} (depende de la categoría o del caso; confirmar)"),
                l.kind,
                json!({ "value": l.value, "others": l.others, "conditional": l.conditional }),
                false,
                l.evidence.clone(),
            );
        } else {
            add(text, l.kind, json!(l.value), blocking, l.evidence.clone());
        }
    }
    for d in x.dates.iter().filter(|d| d.kind == "application_window") {
        add("Ventana de postulación".to_string(), "deadline", json!({ "start": d.start, "end": d.end }), true, d.evidence.clone());
    }
    for d in x.documents.iter().filter(|d| d.source == "eligibility" || d.source == "review") {
        add(format!("Documento: {}", d.text), "required_document", json!({ "name": d.text }), true, d.evidence.clone());
    }
    for d in x.documents.iter().filter(|d| d.source == "annex") {
        add(format!("Documento {}: {}", d.number.unwrap_or(0), d.text), "required_document", json!({ "number": d.number, "name": d.text, "stage": null }), false, d.evidence.clone());
    }
    for group in ["inversion", "operativo", "administrativo", "otro"] {
        let concepts: Vec<&str> = x.funded_concepts.iter().filter(|c| c.group == group).map(|c| c.text.as_str()).collect();
        if let Some(first) = x.funded_concepts.iter().find(|c| c.group == group) {
            add(
                format!("Rubro financiable ({group}): {} conceptos", concepts.len()),
                "allowed_category",
                json!({ "group": group, "concepts": concepts }),
                false,
                Evidence { page: first.page, snippet: format!("RUBRO: {group}") },
            );
        }
    }
    // criteria that no typed rule covers stay as manual requirements for a person to mark
    for c in &x.eligibility {
        let typed = re!(r"(?i)sin\s+fines\s+de\s+lucro\s*\(|al\s+menos\s+\S+\s+a[nñ]os\s+de\s+operaci").is_match(&c.text)
            || c.text.ends_with(':');
        if !typed {
            add(c.text.clone(), "manual", json!(null), true, c.evidence.clone());
        }
    }
    out
}

// ------------------------------------------------------------------ criteria without bullets

/// Criteria written as plain lines under a heading ("CRITERIOS OBLIGATORIOS QUE CUMPLIR…"), as in
/// calls whose bullet symbols were drawn as images and left no character in the text. Lines are
/// grouped into sentences: an item ends where a line ends in a period.
fn find_unbulleted_criteria(pages: &[FlatPage]) -> Vec<TextItem> {
    let mut out = Vec::new();
    for p in pages {
        let lines: Vec<&str> = p.clean.lines().map(str::trim).collect();
        let Some(start) = lines.iter().position(|l| is_heading(l) && l.to_uppercase().contains("CRITERIOS")) else { continue };
        let mut i = start;
        // the heading itself wraps over several lines
        while i < lines.len() && (lines[i].is_empty() || is_heading(lines[i])) {
            i += 1;
        }
        let mut current = String::new();
        for l in &lines[i..] {
            if l.starts_with("NOTA") || (is_heading(l) && current.is_empty()) {
                break;
            }
            if l.is_empty() || l.chars().all(|c| c.is_ascii_digit()) {
                continue;
            }
            if !current.is_empty() {
                current.push(' ');
            }
            current.push_str(l);
            if l.ends_with('.') || l.ends_with(':') {
                let text = flat(&current);
                if (25..=500).contains(&text.chars().count()) {
                    out.push(TextItem { evidence: Evidence { page: p.number, snippet: text.clone() }, text });
                }
                current.clear();
            }
        }
    }
    out
}

// ------------------------------------------------------------------ repeated pages

fn shingles(flat: &str) -> std::collections::BTreeSet<String> {
    let words: Vec<String> = flat
        .split_whitespace()
        .map(|w| w.chars().filter(|c| c.is_alphanumeric()).collect::<String>().to_lowercase())
        .filter(|w| !w.is_empty())
        .collect();
    words.windows(6).map(|w| w.join(" ")).collect()
}

/// `(page, repeats_page)`: a page whose six-word phrases are 85 % contained in an earlier page.
/// Comparing phrases, not lines, makes it indifferent to how the line breaks fall.
fn find_duplicate_pages(pages: &[FlatPage]) -> Vec<(usize, usize)> {
    let sets: Vec<_> = pages.iter().map(|p| shingles(&p.flat)).collect();
    let mut dups: Vec<(usize, usize)> = Vec::new();
    for j in 1..pages.len() {
        if sets[j].len() < 30 {
            continue;
        }
        for i in 0..j {
            if dups.iter().any(|(d, _)| *d == pages[i].number) {
                continue;
            }
            let shared = sets[j].intersection(&sets[i]).count();
            if shared * 100 >= sets[j].len() * 85 {
                dups.push((pages[j].number, pages[i].number));
                break;
            }
        }
    }
    dups
}

// ------------------------------------------------------------------ document kind

/// What the document is, from counting words and line shapes. `typed_signals` is how many limits,
/// dates and criteria the rules already found: a text that yields several of them is a call even
/// when it is too short to use the word counts.
fn classify(pages: &[FlatPage], typed_signals: usize) -> (DocKind, Vec<String>) {
    let all = pages.iter().map(|p| p.flat.as_str()).collect::<Vec<_>>().join(" ").to_lowercase();
    let funding = re!(r"donativ|rubros?\b|financiamiento|apoyos?\s+econ[oó]mic|\bmonto\b").find_iter(&all).count();
    let governance = re!(r"candidat|votaci[oó]n|\bvoto\b|consej|elecci[oó]n|sufragio|padr[oó]n").find_iter(&all).count();
    let lines: Vec<&str> = pages.iter().flat_map(|p| p.clean.lines()).map(str::trim).filter(|l| !l.is_empty()).collect();
    let indicator_lines = lines.iter().filter(|l| re!(r"^(?:Porcentaje|Promedio|#)\s").is_match(l)).count();
    let rubro_lines = lines.iter().filter(|l| l.starts_with("RUBRO:")).count();
    let first = pages.first().map(|p| p.flat.chars().take(300).collect::<String>().to_lowercase()).unwrap_or_default();
    let reasons = vec![format!(
        "términos de financiamiento: {funding}; de gobierno o elección: {governance}; líneas de indicador: {indicator_lines} de {}; líneas «RUBRO:»: {rubro_lines}; datos tipados hallados: {typed_signals}",
        lines.len()
    )];
    let kind = if pages.is_empty() {
        DocKind::Unknown
    } else if rubro_lines >= 2 && pages.len() <= 3 {
        DocKind::RubrosAnnex
    } else if pages.len() <= 4 && !lines.is_empty() && indicator_lines * 4 >= lines.len() {
        DocKind::Indicators
    } else if first.contains("guía") || first.contains("guia") || first.contains("instructivo") {
        // checked before "governance": a guide to filling in a form talks about boards and
        // councils (the people who sign) without being an election notice
        DocKind::Guide
    } else if governance >= 8 && governance * 2 > funding * 3 {
        DocKind::GovernanceNotice
    } else if funding >= 15 || typed_signals >= 4 {
        DocKind::FundingCall
    } else {
        DocKind::Unknown
    };
    (kind, reasons)
}

/// Extracts a call from its pages' text.
/// The dates the rules read in one piece of text (a candidate of the rail), as if it were its own
/// page. `year` is the document's year, for dates written without one.
pub(crate) fn dates_in(text: &str, page: usize, year: Option<u32>) -> Vec<DateItem> {
    let clean = clean_text(text);
    find_dates(&[FlatPage { number: page, flat: flat(&clean), clean }], year)
}

/// Like `dates_in`, also reading "15 Julio": for a line a model has said holds a date.
pub(crate) fn dates_in_loose(text: &str, page: usize, year: Option<u32>) -> Vec<DateItem> {
    let clean = clean_text(text);
    find_dates_with(&[FlatPage { number: page, flat: flat(&clean), clean }], year, true)
}

/// The limits the rules read in one piece of text.
pub(crate) fn limits_in(text: &str, page: usize) -> Vec<Limit> {
    let clean = clean_text(text);
    find_limits(&[FlatPage { number: page, flat: flat(&clean), clean }])
}

pub fn extract_call(pages: &[PageText]) -> CallExtraction {
    let mut repaired_pages = Vec::new();
    let mut low_quality_pages = Vec::new();
    let fp: Vec<FlatPage> = pages
        .iter()
        .map(|p| {
            let (clean, repaired) = clean_text_report(&p.text);
            if repaired {
                repaired_pages.push(p.number);
            }
            if is_low_quality(&clean) {
                low_quality_pages.push(p.number);
            }
            FlatPage { number: p.number, flat: flat(&clean), clean }
        })
        .collect();

    // some PDFs space the characters of a title: "20 24"
    let year = fp
        .first()
        .and_then(|p| re!(r"\b(20)\s?(\d{2})\b").captures(&p.flat).and_then(|c| format!("{}{}", &c[1], &c[2]).parse().ok()))
        .or_else(|| fp.iter().find_map(|p| re!(r"(?i)convocatoria\s+(20\d{2})").captures(&p.flat).and_then(|c| c[1].parse().ok())));
    let title = fp.first().map(|p| p.flat.clone()).filter(|t| !t.is_empty() && t.chars().count() <= 120);

    // some PDFs carry the whole text again on every page; read each text once
    let duplicate_pages = find_duplicate_pages(&fp);
    let all_pages = fp.clone();
    let fp: Vec<FlatPage> = fp.into_iter().filter(|p| !duplicate_pages.iter().any(|(d, _)| *d == p.number)).collect();

    let mut needs_review: Vec<Review> = Vec::new();
    if !duplicate_pages.is_empty() {
        let dup: Vec<usize> = duplicate_pages.iter().map(|(d, _)| *d).collect();
        needs_review.push(Review {
            reason: format!("Las páginas {dup:?} repiten el texto de otra página (el PDF arrastra el mismo contenido); se leyó una sola vez"),
            page: duplicate_pages[0].0,
            snippet: String::new(),
        });
    }
    // The criteria live on the page that states the minimum years or the legal figures, even when
    // that page has no title about participation (found on a held-out call: page 8 was five bullets).
    let limits = find_limits(&fp);
    let legal_figures = find_legal_figures(&fp);
    let rule_pages: Vec<usize> = limits
        .iter()
        .filter(|l| l.kind == "min_operating_years")
        .map(|l| l.evidence.page)
        .chain(legal_figures.iter().map(|l| l.evidence.page))
        .collect();
    let (mut eligibility, eligibility_docs) = find_eligibility(&fp, &rule_pages);
    if eligibility.is_empty() {
        eligibility = find_unbulleted_criteria(&fp);
    }
    let mut documents = eligibility_docs;
    documents.extend(find_annex_documents(&fp, &mut needs_review));
    let (links, emails) = find_links(&fp);
    let indicators = find_indicators(&fp, &mut needs_review);
    let dates = find_dates(&fp, year);
    let (doc_kind, doc_kind_reasons) = classify(&all_pages, limits.len() + dates.len() + eligibility.len());

    let mut x = CallExtraction {
        schema: SCHEMA,
        pages: pages.len(),
        doc_kind,
        doc_kind_reasons,
        duplicate_pages,
        pages_without_text: Vec::new(),
        needs_ocr: false,
        low_quality_pages,
        repaired_pages,
        title,
        year,
        limits,
        legal_figures,
        dates,
        eligibility,
        documents,
        funded_concepts: find_concepts(&fp),
        links,
        emails,
        indicators,
        external: find_external(&fp),
        page_tags: tag_pages(&all_pages),
        needs_review,
        requirements: Vec::new(),
    };
    for e in &x.external {
        x.needs_review.push(Review {
            reason: format!("El valor «{}» no está en el documento: lo da la {}; debe capturarse aparte", e.name, e.lives_in),
            page: e.evidence.page,
            snippet: e.evidence.snippet.clone(),
        });
    }
    for l in &x.limits {
        if l.conditional || !l.others.is_empty() {
            let detail = if l.others.is_empty() {
                "el texto lo liga a una categoría o a un caso".to_string()
            } else {
                format!("el documento da también otros valores: {:?}", l.others)
            };
            x.needs_review.push(Review {
                reason: format!("«{}» = {} no es un valor único: {detail}; una persona debe confirmar a qué aplica", l.kind, l.value),
                page: l.evidence.page,
                snippet: l.evidence.snippet.clone(),
            });
        }
    }
    // dates with no year that sit together come from a calendar drawn as a graphic: the text
    // gives the dates but not which stage each belongs to
    let mut inferred_pages: Vec<usize> = x.dates.iter().filter(|d| d.year_inferred).map(|d| d.evidence.page).collect();
    inferred_pages.sort_unstable();
    let mut seen: Vec<usize> = Vec::new();
    for page in inferred_pages.iter().copied() {
        if seen.contains(&page) {
            continue;
        }
        seen.push(page);
        let n = inferred_pages.iter().filter(|p| **p == page).count();
        if n >= 2 {
            x.needs_review.push(Review {
                reason: format!("Calendario probablemente en gráfico: {n} fechas sin año salieron juntas; el texto no dice a qué etapa corresponde cada una, una persona debe confirmarlo"),
                page,
                snippet: String::new(),
            });
        }
    }
    // requirements only make sense for a call (or the list of fundable concepts on its own)
    match x.doc_kind {
        DocKind::GovernanceNotice => x.needs_review.push(Review {
            reason: "Este documento no es una convocatoria de donativos (es un aviso oficial de otro tipo): no se generan requisitos".into(),
            page: 1,
            snippet: String::new(),
        }),
        DocKind::Guide => x.needs_review.push(Review {
            reason: "Es una guía: sirve como material para que la IA lo lea por secciones, no se extraen requisitos".into(),
            page: 1,
            snippet: String::new(),
        }),
        DocKind::Indicators => {}
        DocKind::FundingCall | DocKind::RubrosAnnex | DocKind::Unknown => x.requirements = requirement_drafts(&x),
    }
    x
}

/// Reads a PDF and extracts the call. A PDF that is mostly images is reported with `needs_ocr`
/// and nothing is extracted from it.
pub fn extract_pdf(path: &Path) -> Result<CallExtraction, PdfError> {
    let pdf = read_pdf(path)?;
    let pages: Vec<PageText> = pdf.pages.iter().enumerate().map(|(i, t)| PageText { number: i + 1, text: t.clone() }).collect();
    let needs_ocr = pdf.pages_without_text.len() * 2 > pdf.pages.len();
    let mut x = if needs_ocr { extract_call(&[]) } else { extract_call(&pages) };
    x.pages = pdf.pages.len();
    x.pages_without_text = pdf.pages_without_text;
    x.needs_ocr = needs_ocr;
    Ok(x)
}

    #[test]
    fn a_24_hour_time_is_read_with_or_without_a_year() {
        // found on a held-out call: "el 10 abril a las 11:00 hrs", with no year, was not read at all
        let d = find_dates(&pages("Se realizará una sesión informativa el 10 de abril de 2027 a las 11:00 hrs."), Some(2027));
        assert_eq!(d.len(), 1, "{d:?}");
        assert_eq!((d[0].start.as_str(), d[0].precision, d[0].kind), ("2027-04-10T11:00", "datetime", "info_session"));
        assert!(!d[0].year_inferred);
        let d = find_dates(&pages("Asiste a la sesión informativa el 10 abril a las 11:00 hrs."), Some(2024));
        assert_eq!(d.len(), 1, "{d:?}");
        assert_eq!((d[0].start.as_str(), d[0].kind, d[0].year_inferred), ("2024-04-10T11:00", "info_session", true));
        // a day and a month with no time and no year is too loose to be a date, and a bad hour is not a time
        assert!(find_dates(&pages("El 10 abril es feriado."), Some(2024)).is_empty());
        assert!(find_dates(&pages("el 10 abril a las 27:00 hrs"), Some(2024)).is_empty());
        // 12-hour times still work
        let d = find_dates(&pages("Sesión el 6 mayo 2027 12:00 p. m. CDMX"), Some(2027));
        assert_eq!(d[0].start, "2027-05-06T12:00");
    }

/// Measures the extractor against a REAL call, comparing it with what a person read by hand.
/// The PDF is not part of the repository: its path comes from the environment.
///   $env:CIMIENTO_CALL_PDF="D:\...\convocatoria.pdf"
///   cargo test measure_real_call -- --ignored --nocapture
#[cfg(test)]
mod real_call {
    use super::*;

    struct Fact {
        name: &'static str,
        found: bool,
        /// A rule must settle it; failing makes the run fail.
        must: bool,
    }

    #[test]
    #[ignore]
    fn measure_real_call() {
        let pdf = std::env::var("CIMIENTO_CALL_PDF").expect("set CIMIENTO_CALL_PDF to a call PDF");
        let x = extract_pdf(Path::new(&pdf)).unwrap();
        let limit = |k: &str| x.limits.iter().find(|l| l.kind == k).map(|l| l.value);
        let date = |kind: &str, start: &str, end: Option<&str>| x.dates.iter().any(|d| d.kind == kind && d.start == start && d.end.as_deref() == end);
        let n = |g: &str| x.funded_concepts.iter().filter(|c| c.group == g).count();
        let figs = x.legal_figures.as_ref().map(|l| l.values.clone()).unwrap_or_default();
        let url = |frag: &str| x.links.iter().any(|l| l.url.contains(frag));
        let annex = x.documents.iter().filter(|d| d.source == "annex").count();
        let listed = x.documents.iter().filter(|d| d.source == "eligibility").count();
        let annex_text = |n: u32| x.documents.iter().find(|d| d.source == "annex" && d.number == Some(n)).map(|d| d.text.clone()).unwrap_or_default();

        // What a person read by hand in the 2026 NMP call (see docs/11).
        let facts = vec![
            Fact { name: "duración máxima = 12 meses", found: limit("max_duration_months") == Some(12.0), must: true },
            Fact { name: "gastos administrativos ≤ 15 %", found: limit("max_admin_percent") == Some(15.0), must: true },
            Fact { name: "mínimo 3 años de operación", found: limit("min_operating_years") == Some(3.0), must: true },
            Fact { name: "figuras AC, IAP, IBP, ABP", found: figs == vec!["AC", "IAP", "IBP", "ABP"], must: true },
            Fact { name: "ventana de postulación 8–15 abril 2026", found: date("application_window", "2026-04-08", Some("2026-04-15")), must: true },
            Fact { name: "sesión informativa 7 abril 12:30", found: date("info_session", "2026-04-07T12:30", None), must: true },
            Fact { name: "sesión informativa 8 abril 10:00", found: date("info_session", "2026-04-08T10:00", None), must: true },
            Fact { name: "entrega de recursos a partir de julio", found: date("disbursement", "2026-07", None), must: true },
            Fact { name: "8 criterios de la página 4", found: x.eligibility.iter().filter(|c| c.evidence.page == 4).count() == 8, must: true },
            Fact { name: "criterio «no se apoyarán organizaciones … religiosos»", found: x.eligibility.iter().any(|c| c.text.contains("No se apoyarán") && c.text.contains("religiosos")), must: true },
            Fact { name: "regla «solo una agenda» (página 8)", found: x.eligibility.iter().any(|c| c.text.contains("una sola agenda") && c.evidence.page == 8), must: true },
            Fact { name: "regla «no se reciben proyectos fuera de fecha» (página 8)", found: x.eligibility.iter().any(|c| c.text.contains("fuera de las fechas")), must: true },
            Fact { name: "11 criterios en total (8 + 3), sin los pasos de procedimiento de la página 9", found: x.eligibility.len() == 11 && !x.eligibility.iter().any(|c| c.evidence.page == 9), must: true },
            Fact { name: "7 documentos listados en los criterios", found: listed == 7, must: true },
            Fact { name: "el 1.º de esos documentos son los estados financieros 2022–2024", found: x.documents.iter().find(|d| d.source == "eligibility").is_some_and(|d| d.text.contains("2022, 2023 y 2024")), must: true },
            Fact { name: "21 documentos numerados del Anexo 1", found: annex == 21, must: true },
            Fact { name: "Anexo 1, n.º 1 = estados financieros", found: annex_text(1).contains("Estados Financieros"), must: true },
            Fact { name: "Anexo 1, n.º 3 = acta constitutiva", found: annex_text(3).contains("Acta constitutiva"), must: true },
            Fact { name: "Anexo 1, n.º 11 = opinión de cumplimiento del SAT", found: annex_text(11).contains("Opinión de cumplimiento del SAT"), must: true },
            Fact { name: "Anexo 1, n.º 21 = comprobante de domicilio", found: annex_text(21).contains("Comprobante de domicilio"), must: true },
            Fact { name: "el Anexo 1 no trae pasos de las etapas (Postulación, Registro…)", found: x.documents.iter().filter(|d| d.source == "annex").all(|d| !["Postulación.", "Registro.", "Entrega de Recursos."].contains(&d.text.as_str())), must: true },
            Fact { name: "sin fechas repetidas", found: x.dates.iter().enumerate().all(|(i, a)| x.dates.iter().skip(i + 1).all(|b| !(a.kind == b.kind && a.start == b.start && a.end == b.end))), must: true },
            Fact { name: "9 conceptos de inversión", found: n("inversion") == 9, must: true },
            Fact { name: "46 conceptos operativos", found: n("operativo") == 46, must: true },
            Fact { name: "7 conceptos administrativos", found: n("administrativo") == 7, must: true },
            Fact { name: "9 enlaces (8 con esquema y 1 «www.» sin esquema)", found: x.links.len() == 9, must: true },
            Fact { name: "enlace sin esquema de la línea ética", found: url("tipsanonimos.com/tulineaetica"), must: true },
            Fact { name: "enlace de sesión completo (reunido en 2 líneas)", found: url("b03b4b99-667c-43cf-b6ea-c7f461e1d86b@f0113a99-d66e-4a33-8db7-618141ec1fd0"), must: true },
            Fact { name: "enlace del formulario", found: url("tfaforms.com/5211814"), must: true },
            Fact { name: "enlace de complementos (ZIP)", found: x.links.iter().any(|l| l.kind == "download" && l.url.contains("Documentos-Convocatoria.zip")), must: true },
            Fact { name: "2 correos", found: x.emails.len() == 2, must: true },
            Fact { name: "el monto se marca como externo (plataforma)", found: x.external.iter().any(|e| e.name == "monto"), must: true },
            Fact { name: "NO inventa montos (el PDF no trae ninguno)", found: limit("max_amount").is_none() && limit("min_amount").is_none(), must: true },
            Fact { name: "marca la tabla de documentos por etapa", found: x.needs_review.iter().any(|r| r.reason.contains("Matriz de documentos por etapa")), must: true },
            Fact { name: "marca la tabla de indicadores", found: x.needs_review.iter().any(|r| r.reason.contains("Tabla de indicadores")), must: true },
            Fact { name: "no pide OCR (el PDF tiene texto)", found: !x.needs_ocr, must: true },
        ];

        println!("\nPDF: {pdf}");
        println!("páginas: {} · sin texto: {:?} · reparadas: {:?} · baja calidad: {:?}", x.pages, x.pages_without_text, x.repaired_pages, x.low_quality_pages);
        println!("\nContra lo leído a mano:");
        for f in &facts {
            println!("  [{}] {}", if f.found { "x" } else { " " }, f.name);
        }
        let got = facts.iter().filter(|f| f.found).count();
        println!("\n{got} de {} hechos de referencia los saca el código por reglas ({:.0} %)", facts.len(), got as f64 * 100.0 / facts.len() as f64);
        println!("\nLo que el código extrajo (cantidades):");
        println!("  límites: {} · fechas: {} · criterios: {} · documentos: {} · conceptos: {} · enlaces: {} · indicadores: {}", x.limits.len(), x.dates.len(), x.eligibility.len(), x.documents.len(), x.funded_concepts.len(), x.links.len(), x.indicators.len());
        println!("  borradores de requisito: {}  ({} por escribir a mano, tipo manual)", x.requirements.len(), x.requirements.iter().filter(|r| r.kind == "manual").count());
        println!("\nLo que NO resuelve por reglas y queda para persona o IA:");
        for r in &x.needs_review {
            println!("  - p.{}: {}", r.page, r.reason);
        }
        let narrative: Vec<usize> = x.page_tags.iter().filter(|p| p.tags.iter().any(|t| ["evaluacion", "estrategias", "poblacion"].contains(t))).map(|p| p.page).collect();
        println!("  - texto narrativo (evaluación, estrategias, población) en las páginas {narrative:?}: la IA lo lee por etiqueta, solo si lo necesita");
        println!("\nVentana de postulación frente a hoy: {:?}", window_status(&x, &crate::ai::settings::today_utc()));
        if let Ok(path) = std::env::var("CIMIENTO_EXTRACT_JSON") {
            std::fs::write(&path, serde_json::to_string_pretty(&x).unwrap()).unwrap();
            println!("JSON completo escrito en {path}");
        }
        let failed: Vec<&str> = facts.iter().filter(|f| f.must && !f.found).map(|f| f.name).collect();
        assert!(failed.is_empty(), "el extractor no saca: {failed:?}");
    }
}

/// Runs the extractor over every PDF in a folder and prints one summary per file, so a new
/// document is seen as it is BEFORE any rule is touched.
///   $env:CIMIENTO_PDF_DIR="D:\...\carpeta"; $env:CIMIENTO_EXTRACT_DIR="D:\...\salida"
///   cargo test survey_pdf_folder -- --ignored --nocapture
#[cfg(test)]
mod survey {
    use super::*;

    #[test]
    #[ignore]
    fn survey_pdf_folder() {
        let dir = std::env::var("CIMIENTO_PDF_DIR").expect("set CIMIENTO_PDF_DIR");
        let out_dir = std::env::var("CIMIENTO_EXTRACT_DIR").ok();
        let mut files: Vec<_> = std::fs::read_dir(&dir)
            .unwrap()
            .filter_map(|e| e.ok().map(|e| e.path()))
            .filter(|p| p.extension().is_some_and(|x| x.eq_ignore_ascii_case("pdf")))
            .collect();
        files.sort();
        for path in files {
            let name = path.file_name().unwrap().to_string_lossy().to_string();
            println!("\n##### {name}");
            let x = match extract_pdf(&path) {
                Ok(x) => x,
                Err(e) => {
                    println!("  NO SE PUDO LEER: {e}");
                    continue;
                }
            };
            println!(
                "  páginas {} · sin texto {:?} · pide OCR: {} · reparadas {:?} · baja calidad {:?}",
                x.pages, x.pages_without_text, x.needs_ocr, x.repaired_pages, x.low_quality_pages
            );
            println!("  TIPO: {:?} — {}", x.doc_kind, x.doc_kind_reasons.join(" "));
            if !x.duplicate_pages.is_empty() {
                println!("  páginas repetidas (página, repite a): {:?}", x.duplicate_pages);
            }
            println!("  título: {:?} · año: {:?}", x.title, x.year);
            println!(
                "  límites {} · fechas {} · criterios {} · documentos {} · conceptos {} · enlaces {} · correos {} · indicadores {} · requisitos {} (manuales {})",
                x.limits.len(), x.dates.len(), x.eligibility.len(), x.documents.len(), x.funded_concepts.len(),
                x.links.len(), x.emails.len(), x.indicators.len(), x.requirements.len(),
                x.requirements.iter().filter(|r| r.kind == "manual").count()
            );
            if !x.limits.is_empty() {
                println!("  límites: {:?}", x.limits.iter().map(|l| (l.kind, l.value)).collect::<Vec<_>>());
            }
            if let Some(l) = &x.legal_figures {
                println!("  figuras: {:?}", l.values);
            }
            for d in x.dates.iter().take(8) {
                println!("  fecha {} {} {:?} (p.{})", d.kind, d.start, d.end, d.evidence.page);
            }
            for r in x.needs_review.iter().take(6) {
                println!("  revisar p.{}: {}", r.page, r.reason.chars().take(110).collect::<String>());
            }
            for e in &x.external {
                println!("  externo: {} en {}", e.name, e.lives_in);
            }
            println!("  ventana frente a hoy: {:?}", window_status(&x, &crate::ai::settings::today_utc()));
            if let Some(o) = &out_dir {
                let stem = path.file_stem().unwrap().to_string_lossy().replace(|c: char| !c.is_ascii_alphanumeric(), "_");
                std::fs::write(format!("{o}/{stem}.json"), serde_json::to_string_pretty(&x).unwrap()).unwrap();
            }
        }
    }
}

#[cfg(test)]
mod exploration {

    /// Dumps the pages `pdf-extract` reads from a PDF, so the extractor is written against the
    /// real text and not against a guess.
    ///   $env:CIMIENTO_CALL_PDF="D:\...\convocatoria.pdf"; $env:CIMIENTO_DUMP="D:\...\paginas.txt"
    ///   cargo test dump_call_pages -- --ignored --nocapture
    #[test]
    #[ignore]
    fn dump_call_pages() {
        let pdf = std::env::var("CIMIENTO_CALL_PDF").expect("set CIMIENTO_CALL_PDF");
        let out = std::env::var("CIMIENTO_DUMP").expect("set CIMIENTO_DUMP");
        let r = read_pdf(Path::new(&pdf)).unwrap();
        let mut s = String::new();
        for (i, p) in r.pages.iter().enumerate() {
            s.push_str(&format!("\n=== PAGINA {} ({} caracteres) ===\n{}", i + 1, p.chars().count(), p));
        }
        std::fs::write(&out, s).unwrap();
        println!("{} páginas; sin texto: {:?}", r.pages.len(), r.pages_without_text);
    }
